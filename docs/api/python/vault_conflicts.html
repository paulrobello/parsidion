vault_conflicts

vault-conflicts: detect contradictions between semantically-similar vault notes.

Companion to vault-merge (which merges near-duplicates). Clusters notes by embedding similarity, then asks the configured prompt AI backend whether any pair in a cluster makes mutually-exclusive claims. Conflicts can be reviewed interactively (curses TUI) or emitted as JSON.

 35def _parse_json_array(text: str) -> list[dict[str, Any]]:
 36    """Extract the first JSON array from *text*; return [] if none/unparseable.
 37
 38    Tolerates markdown fences and surrounding prose from an LLM.
 39    """
 40    if not text:
 41        return []
 42    stripped = text.strip()
 43    if stripped.startswith("```"):
 44        # Remove a leading ```json / ``` fence and trailing fence.
 45        stripped = re.sub(r"^```[a-zA-Z]*\n?", "", stripped)
 46        stripped = re.sub(r"\n?```$", "", stripped)
 47    start = stripped.find("[")
 48    end = stripped.rfind("]")
 49    if start == -1 or end == -1 or end < start:
 50        return []
 51    try:
 52        parsed = json.loads(stripped[start : end + 1])
 53    except json.JSONDecodeError:
 54        return []
 55    if not isinstance(parsed, list):
 56        return []
 57    return [item for item in parsed if isinstance(item, dict)]

 60def _cosine_similarity(a: list[float], b: list[float]) -> float:
 61    """Cosine similarity of two equal-length float vectors; 0.0 for zero vectors."""
 62    dot = 0.0
 63    na = 0.0
 64    nb = 0.0
 65    for x, y in zip(a, b, strict=True):
 66        dot += x * y
 67        na += x * x
 68        nb += y * y
 69    if na == 0.0 or nb == 0.0:
 70        return 0.0
 71    return dot / ((na**0.5) * (nb**0.5))

 74def _group_clusters(n: int, pairs: list[tuple[int, int]]) -> list[list[int]]:
 75    """Union-find over *pairs*; return ONLY clusters with >= 2 members."""
 76    parent = list(range(n))
 77
 78    def find(x: int) -> int:
 79        while parent[x] != x:
 80            parent[x] = parent[parent[x]]
 81            x = parent[x]
 82        return x
 83
 84    for i, j in pairs:
 85        ri, rj = find(i), find(j)
 86        if ri != rj:
 87            parent[ri] = rj
 88
 89    groups: dict[int, list[int]] = defaultdict(list)
 90    for idx in range(n):
 91        groups[find(idx)].append(idx)
 92    return [sorted(members) for members in groups.values() if len(members) >= 2]

101def _load_embeddings(
102    vault: Path,
103) -> tuple[list[dict[str, str]], list[list[float]]]:
104    """Load note_embeddings rows + unpacked float32 vectors.
105
106    Returns parallel lists of (records, vectors). Empty if the DB or table is
107    missing (the caller should print a helpful message).
108    """
109    import sqlite3
110
111    db_path = vault_common.get_embeddings_db_path(vault)
112    if not db_path.exists():
113        return [], []
114    conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
115    try:
116        rows = conn.execute(
117            "SELECT stem, path, folder, title, tags, embedding FROM note_embeddings"
118        ).fetchall()
119    except sqlite3.Error:
120        return [], []
121    finally:
122        conn.close()
123
124    records: list[dict[str, str]] = []
125    vectors: list[list[float]] = []
126    blobs = []
127    global _skipped_outside_warned
128    for stem, path, folder, title, tags, blob in rows:
129        if _is_excluded(path):
130            continue
131        # SEC-020: embeddings.db is a tamperable store; a row whose path
132        # resolves outside the vault must not reach _read_body or the
133        # contradiction prompt. Mirrors _paths_from_rows in vault_index.
134        if not is_path_inside_vault(Path(path), vault):
135            if not _skipped_outside_warned:
136                _skipped_outside_warned = True
137                print(
138                    "vault-conflicts: skipped note rows whose paths resolve "
139                    "outside the vault (tampered embeddings.db?); SEC-020",
140                    file=sys.stderr,
141                )
142            continue
143        records.append(
144            {
145                "stem": stem,
146                "path": path,
147                "folder": folder,
148                "title": title,
149                "tags": tags,
150            }
151        )
152        blobs.append(blob)
153    if not blobs:
154        return [], []
155    dim = len(blobs[0]) // 4
156    vectors = [list(struct.unpack(f"{dim}f", b)) for b in blobs]
157    return records, vectors

160def _python_pairwise(
161    vectors: list[list[float]], threshold: float
162) -> list[tuple[int, int]]:
163    """Pure-Python O(n^2) pairwise cosine — fallback when sqlite-vec is absent.
164
165    Correct but slow at scale (~minutes for thousands of notes); used in the
166    dev/test environment and minimal installs without the ``[tools]`` extras.
167    """
168    pairs: list[tuple[int, int]] = []
169    n = len(vectors)
170    for i in range(n):
171        vi = vectors[i]
172        for j in range(i + 1, n):
173            if _cosine_similarity(vi, vectors[j]) >= threshold:
174                pairs.append((i, j))
175    return pairs

178def _sqlite_pairwise(
179    vault: Path, stem_to_idx: dict[str, int], threshold: float
180) -> list[tuple[int, int]] | None:
181    """Fast pairwise cosine via sqlite-vec's C-level ``vec_distance_cosine``.
182
183    Returns ``None`` when sqlite-vec is unavailable or the query fails, so the
184    caller falls back to pure Python. ``cosine similarity >= threshold`` is
185    equivalent to ``vec_distance_cosine <= (1 - threshold)``.
186    """
187    try:
188        import sqlite3
189
190        import sqlite_vec  # type: ignore[import-untyped]
191    except ImportError:
192        return None
193    db_path = vault_common.get_embeddings_db_path(vault)
194    if not db_path.exists():
195        return None
196    max_dist = 1.0 - threshold
197    conn = None
198    try:
199        conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
200        conn.enable_load_extension(True)
201        sqlite_vec.load(conn)
202        conn.enable_load_extension(False)
203        # Upper-triangle self-join: each pair once, no self-pairs.
204        rows = conn.execute(
205            """
206            SELECT a.stem, b.stem
207            FROM note_embeddings a
208            JOIN note_embeddings b ON a.rowid < b.rowid
209            WHERE vec_distance_cosine(a.embedding, b.embedding) <= ?
210            """,
211            (max_dist,),
212        ).fetchall()
213    except sqlite3.Error:
214        return None
215    finally:
216        if conn is not None:
217            conn.close()
218    pairs: list[tuple[int, int]] = []
219    for a, b in rows:
220        ia = stem_to_idx.get(a)
221        ib = stem_to_idx.get(b)
222        if ia is not None and ib is not None:
223            pairs.append((ia, ib))
224    return pairs

227def find_candidate_clusters(
228    vault: Path, threshold: float = _DEFAULT_TOPIC_THRESHOLD, top: int = _DEFAULT_TOP
229) -> list[list[dict[str, str]]]:
230    """Cluster semantically-similar notes; return clusters with >= 2 members.
231
232    Uses sqlite-vec's C-level cosine when available (fast at scale); falls back
233    to a pure-Python O(n^2) scan otherwise.
234    """
235    records, vectors = _load_embeddings(vault)
236    n = len(records)
237    if n < 2:
238        return []
239    stem_to_idx = {rec["stem"]: i for i, rec in enumerate(records)}
240    pairs = _sqlite_pairwise(vault, stem_to_idx, threshold)
241    if pairs is None:
242        pairs = _python_pairwise(vectors, threshold)
243    clusters: list[list[dict[str, str]]] = []
244    for member_indices in _group_clusters(n, pairs):
245        cluster = [records[idx] for idx in member_indices[:top]]
246        if len(cluster) >= 2:
247            clusters.append(cluster)
248    return clusters

251def _read_body(path: str) -> str:
252    """Read a note's body (after frontmatter), truncated for prompt size."""
253    try:
254        text = Path(path).read_text(encoding="utf-8")
255    except OSError:
256        return ""
257    # Drop frontmatter block if present.
258    if text.startswith("---"):
259        end = text.find("\n---", 3)
260        if end != -1:
261            text = text[end + 4 :]
262    return text.strip()[:2000]

265def _build_prompt(records: list[dict[str, str]]) -> str:
266    """Build the contradiction-detection prompt with note bodies inlined."""
267    blocks: list[str] = []
268    for rec in records:

279def _detect_contradictions(
280    records: list[dict[str, str]], vault: Path, no_ai: bool = False
281) -> list[dict[str, Any]]:
282    """Ask the AI backend to find contradictions within a cluster of notes.
283
284    Returns a list of conflict dicts (parsed from the AI's JSON array).
285    """
286    if no_ai or len(records) < 2:
287        return []
288    import ai_backend
289
290    prompt = _build_prompt(records)
291    raw = ai_backend.run_ai_prompt(
292        prompt,
293        model_tier="large",
294        timeout=_DEFAULT_AI_TIMEOUT,
295        purpose="vault-conflicts",
296        vault=vault,
297    )
298    if not raw:
299        return []
300    conflicts = _parse_json_array(raw)
301    # Keep only entries that name two distinct known stems in this cluster.
302    known = {rec["stem"] for rec in records}
303    return [
304        c
305        for c in conflicts
306        if c.get("a") in known and c.get("b") in known and c.get("a") != c.get("b")
307    ]

314def write_conflict_report(conflicts: list[dict[str, Any]], vault: Path) -> None:
315    """Atomically write the conflict report (flock-protected).
316
317    SEC-033: the lock is on the DESTINATION path. Locking the tmp file (the
318    previous behavior) excluded nobody — each run creates a fresh tmp, so two
319    concurrent writers interleaved and the last replace won arbitrarily.
320    Holding the destination's flock across tmp-write + replace serializes
321    writers on the real path.
322    """
323    dest = _report_path(vault)
324    dest.parent.mkdir(parents=True, exist_ok=True)
325    tmp = dest.with_suffix(".json.tmp")
326    with open(dest, "a", encoding="utf-8") as lock_fh:
327        vault_common.flock_exclusive(lock_fh)
328        tmp.write_text(json.dumps(conflicts, indent=2) + "\n", encoding="utf-8")
329        tmp.replace(dest)

344def _apply_resolution(conflict: dict[str, Any], choice: str) -> str:
345    """Return a short description of the applied resolution.
346
347    The TUI (Task 4.6) collects *choice* ∈ {keep_a, keep_b, merge, skip} and
348    calls this. Actual note mutation (deleting/merging files) is intentionally
349    NOT done here — this version records the decision; users perform the edit
350    in their editor or via vault-merge. (Keeps vault-conflicts read-only-safe.)
351    """
352    a = conflict.get("a", "?")
353    b = conflict.get("b", "?")
354    if choice == "keep_a":
355        return f"keep {a}; review {b} for staleness"
356    if choice == "keep_b":
357        return f"keep {b}; review {a} for staleness"
358    if choice == "merge":
359        return f"merge {a} + {b} via vault-merge"
360    return "skipped"

363def _run_scan(
364    vault: Path, threshold: float, top: int, no_ai: bool = False
365) -> list[dict[str, Any]]:
366    """Run the full detect pipeline and persist the report. Returns conflicts."""
367    clusters = find_candidate_clusters(vault, threshold=threshold, top=top)
368    all_conflicts: list[dict[str, Any]] = []
369    for cluster in clusters:
370        members = cluster[:_DEFAULT_MAX_CLUSTER]
371        all_conflicts.extend(_detect_contradictions(members, vault, no_ai=no_ai))
372    write_conflict_report(all_conflicts, vault)
373    return all_conflicts

376def _run_tui(conflicts: list[dict[str, Any]], vault: Path) -> None:  # pragma: no cover
377    """Interactive curses walkthrough (mirrors vault_review._show_popup).
378
379    ARC-013: the loop machinery lives in ``vault_tui.run_list_view``; the
380    selected "row" renders that conflict's A/B block (non-selected rows stay
381    blank, preserving the one-conflict-at-a-time view). For each conflict:
382    show a_says vs b_says, collect a choice (a=keep_a, b=keep_b, m=merge,
383    s=skip, q=quit). Decisions are collected during the curses loop and
384    printed AFTER the wrapper returns (curses owns the screen while active,
385    so stdout is not visible mid-loop).
386    """
387    import curses
388
389    from vault_tui import run_list_view
390
391    decisions: list[str] = []
392    mapping = {
393        ord("a"): "keep_a",
394        ord("b"): "keep_b",
395        ord("m"): "merge",
396        ord("s"): "skip",
397    }
398
399    def _render_conflict(stdscr, c, y, is_selected, w) -> None:
400        if not is_selected:
401            return
402        stdscr.addstr(y, 2, f"[A] {c.get('a')}: {c.get('a_says', '')}"[: w - 1])
403        stdscr.addstr(y + 1, 2, f"[B] {c.get('b')}: {c.get('b_says', '')}"[: w - 1])
404        stdscr.addstr(y + 3, 2, "a=keep A  b=keep B  m=merge  s=skip  q=quit")
405
406    def _on_key(key: int, selected: int) -> str | int | None:
407        if key in (ord("q"), 27):  # q or ESC
408            return "quit"
409        choice = mapping.get(key)
410        if choice:
411            decisions.append(_apply_resolution(conflicts[selected], choice))
412            # Advance to the next conflict; deciding the last one exits
413            # (the original walkthrough ended when idx passed the end).
414            if selected + 1 >= len(conflicts):
415                return "quit"
416            return selected + 1
417        return None
418
419    def _loop(stdscr: Any) -> None:
420        run_list_view(
421            stdscr,
422            conflicts,
423            _render_conflict,
424            _on_key,
425            title=lambda sel: f"Conflict {sel + 1}/{len(conflicts)}",
426        )
427
428    try:
429        curses.wrapper(_loop)
430    except Exception:  # noqa: BLE001
431        print(
432            "Terminal does not support curses; use --scan-only or --json.",
433            file=sys.stderr,
434        )
435        return
436
437    for line in decisions:
438        print(line)
