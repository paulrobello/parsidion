core.vault_schema

Typed schema for the Parsidion vault config (ENH-014).

Single source of truth for the config schema: section names, key names, and the allowed Python types for each key. Each top-level config section is a dataclass whose field annotations reproduce the allowed-type tuples the hand-maintained _CONFIG_SCHEMA literal in vault_config previously encoded by hand. schema_dict() derives that structure from the annotations so validation can consume one source of truth.

Stdlib-only (dataclasses + typing). Values pass through unchanged -- no coercion, no canonical defaults invented. A field value of None means "absent from the file", mirroring the dict load_config() returns. Validation (unknown keys, type mismatches) stays warn-only and lives in vault_config.validate_config(); this module never raises on shape.

The dataclasses deliberately default every field to None even when the field's allowed-type tuple does not include NoneType (e.g. backend: str defaulting to None). The annotation says what config.yaml values are valid; the default says what an unset field holds. validate_config already treats an explicit None value as "skip the type check", so the two notions are consistent.

263@dataclass
264class VaultAppConfig:
265    """Typed view of the parsed vault config.
266
267    Every section is always present: an absent (or non-mapping) section in
268    the file yields an empty instance of its dataclass (all fields ``None``),
269    so consumers can chain ``cfg.session_start_hook.ai_model`` without
270    None-guarding the section itself. Whether a section was *explicitly* set
271    in the file is not distinguishable from this view (check the dict from
272    :func:`load_config` if you need that); individual field values are
273    ``None`` for keys absent from the file. Construct via :meth:`from_dict`
274    or :func:`vault_config.load_typed_config`; the dataclasses carry no
275    validation -- that remains :func:`vault_config.validate_config`'s job.
276    """
277
278    # ``default_factory=lambda: X()`` (rather than ``= X``) sidesteps a pyright
279    # overload-resolution hiccup that arises under ``from __future__ import
280    # annotations`` when the factory is a dataclass whose own fields use the
281    # ``attr: str = None`` pattern the module-level reportAssignmentType
282    # suppression covers. The lambda resolves to ``Callable[[], X]`` directly.
283    ai: AIConfig = field(default_factory=lambda: AIConfig())
284    ai_models: AIModelsConfig = field(default_factory=lambda: AIModelsConfig())
285    codex_cli: CodexCliConfig = field(default_factory=lambda: CodexCliConfig())
286    session_start_hook: SessionStartHookConfig = field(
287        default_factory=lambda: SessionStartHookConfig()
288    )
289    session_stop_hook: SessionStopHookConfig = field(
290        default_factory=lambda: SessionStopHookConfig()
291    )
292    subagent_stop_hook: SubagentStopHookConfig = field(
293        default_factory=lambda: SubagentStopHookConfig()
294    )
295    pre_compact_hook: PreCompactHookConfig = field(
296        default_factory=lambda: PreCompactHookConfig()
297    )
298    summarizer: SummarizerConfig = field(default_factory=lambda: SummarizerConfig())
299    embeddings: EmbeddingsConfig = field(default_factory=lambda: EmbeddingsConfig())
300    par_mem: ParMemConfig = field(default_factory=lambda: ParMemConfig())
301    search: SearchConfig = field(default_factory=lambda: SearchConfig())
302    anthropic_env: AnthropicEnvConfig = field(
303        default_factory=lambda: AnthropicEnvConfig()
304    )
305    git: GitConfig = field(default_factory=lambda: GitConfig())
306    defaults: DefaultsConfig = field(default_factory=lambda: DefaultsConfig())
307    event_log: EventLogConfig = field(default_factory=lambda: EventLogConfig())
308    adaptive_context: AdaptiveContextConfig = field(
309        default_factory=lambda: AdaptiveContextConfig()
310    )
311    vault: VaultSectionConfig = field(default_factory=lambda: VaultSectionConfig())
312    adapters: AdaptersConfig = field(default_factory=lambda: AdaptersConfig())

314    @classmethod
315    def from_dict(cls, parsed: dict[str, Any]) -> VaultAppConfig:
316        """Map a parsed config dict onto the section dataclasses.
317
318        Values pass through unchanged -- no coercion. Absent keys default to
319        ``None`` on the section dataclass. Unknown sections/keys and non-mapping
320        section values are skipped silently; :func:`vault_config.validate_config`
321        is responsible for warning about them.
322
323        Args:
324            parsed: The dict returned by :func:`vault_config.load_config`.
325
326        Returns:
327            A populated :class:`VaultAppConfig`. Returns an instance whose
328            sections are all empty (every field ``None``) when *parsed* is not
329            a dict.
330        """
331        if not isinstance(parsed, dict):
332            return cls()
333
334        hints = get_type_hints(cls)
335        kwargs: dict[str, Any] = {}
336        for f in fields(cls):
337            section_cls = _unwrap_optional(hints[f.name])
338            # ``is_dataclass`` alone narrows ``Any`` to the instance protocol
339            # (not callable); gating on ``isinstance(..., type)`` first lets
340            # pyright see a class we can instantiate.
341            if not (isinstance(section_cls, type) and is_dataclass(section_cls)):
342                continue
343            raw_section = parsed.get(f.name)
344            if not isinstance(raw_section, dict):
345                continue
346            sub_kwargs: dict[str, Any] = {}
347            for sub_f in fields(section_cls):
348                if sub_f.name in raw_section:
349                    sub_kwargs[sub_f.name] = raw_section[sub_f.name]
350            kwargs[f.name] = section_cls(**sub_kwargs)
351        return cls(**kwargs)

359def schema_dict() -> dict[str, dict[str, tuple[type, ...]]]:
360    """Derive the ``section -> key -> allowed-types`` schema from annotations.
361
362    The structure mirrors the hand-maintained ``_CONFIG_SCHEMA`` literal:
363    each section dataclass field's annotation becomes an allowed-types tuple.
364    A bare type (e.g. ``str``) yields ``(str,)``; a union (e.g. ``int | float``)
365    yields ``(int, float)`` preserving annotation order; ``None`` in a union
366    maps to ``type(None)`` (``NoneType``), matching the convention
367    ``validate_config`` already uses to filter None out of warning messages.
368    """
369    result: dict[str, dict[str, tuple[type, ...]]] = {}
370    top_hints = get_type_hints(VaultAppConfig)
371    for f in fields(VaultAppConfig):
372        section_cls = _unwrap_optional(top_hints[f.name])
373        if not (isinstance(section_cls, type) and is_dataclass(section_cls)):
374            continue
375        section_hints = get_type_hints(section_cls)
376        section_schema: dict[str, tuple[type, ...]] = {}
377        for sub_f in fields(section_cls):
378            section_schema[sub_f.name] = _annotation_to_allowed_types(
379                section_hints[sub_f.name]
380            )
381        result[f.name] = section_schema
382    return result

385def _unwrap_optional(tp: Any) -> Any:
386    """Return the single non-``None`` arg of an Optional union, else *tp*.
387
388    ``ai: AIConfig | None`` resolves to a union whose ``__args__`` are
389    ``(AIConfig, NoneType)``; this returns ``AIConfig`` so callers can inspect
390    the section dataclass directly. Unions with multiple non-None args (none
391    here today, but defensively) and non-union types are returned unchanged.
392    """
393    args = getattr(tp, "__args__", None)
394    if args is None:
395        return tp
396    non_none = [a for a in args if a is not type(None)]
397    if len(non_none) == 1:
398        return non_none[0]
399    return tp

402def _annotation_to_allowed_types(ann: Any) -> tuple[type, ...]:
403    """Turn a resolved annotation into the allowed-types tuple.
404
405    A bare type has no ``__args__`` and yields a one-tuple. A union yields
406    its ``__args__`` as a tuple, preserving source order (e.g.
407    ``float | int`` -> ``(float, int)``). ``None`` in a PEP 604 union arrives
408    as ``NoneType`` in ``__args__``, which IS ``type(None)`` -- the form
409    ``_CONFIG_SCHEMA`` used and ``validate_config``'s message filter expects.
410    """
411    args = getattr(ann, "__args__", None)
412    if args is None:
413        return (ann,)
414    return tuple(args)
