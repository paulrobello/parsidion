121def _parse_config_yaml(text: str) -> dict[str, Any]:
122    """Parse a simple YAML config with limited nesting.
123
124    Handles top-level scalars, section dicts, and one additional mapping level
125    for config sections such as ``ai_models.codex.small``::
126
127        top_key: value
128        section:
129          nested_key: value
130          nested_section:
131            leaf_key: value
132    """
133    result: dict[str, Any] = {}
134    current_section: str | None = None
135    current_nested_key: str | None = None
136    current_nested_indent = 0
137    # Indent of leaf keys inside the current nested dict; anything indented
138    # deeper is a 3rd nesting level, which this parser does not support.
139    current_nested_leaf_indent: int | None = None
140
141    for lineno, line in enumerate(text.splitlines(), 1):
142        stripped = line.strip()
143        if not stripped or stripped.startswith("#"):
144            continue
145
146        indent = len(line) - len(line.lstrip())
147        key_value = split_key_value(stripped)
148        if key_value is None:
149            print(
150                f"vault_config: ignoring unparsable config line: {stripped!r}",
151                file=sys.stderr,
152            )
153            continue
154
155        key, value_str = key_value
156
157        if not key:
158            print(
159                f"vault_config: ignoring config line with empty key: {stripped!r}",
160                file=sys.stderr,
161            )
162            continue
163
164        if indent == 0:
165            if not value_str:
166                # Section header -- start collecting nested keys
167                current_section = key
168                current_nested_key = None
169                current_nested_indent = 0
170                current_nested_leaf_indent = None
171                result[key] = {}
172            else:
173                value_str = strip_inline_comment(value_str)
174                result[key] = parse_scalar(value_str)
175                current_section = None
176                current_nested_key = None
177                current_nested_indent = 0
178                current_nested_leaf_indent = None
179        elif current_section is not None and indent > 0:
180            value_str = strip_inline_comment(value_str)
181            section = result.get(current_section)
182            if not isinstance(section, dict):
183                continue
184
185            if (
186                current_nested_key is not None
187                and indent > current_nested_indent
188                and isinstance(section.get(current_nested_key), dict)
189            ):
190                nested = section[current_nested_key]
191                if isinstance(nested, dict):
192                    if current_nested_leaf_indent is None:
193                        current_nested_leaf_indent = indent
194                    if not value_str or indent > current_nested_leaf_indent:
195                        # A 3rd nesting level (a sub-section header inside a
196                        # nested dict, or a key indented deeper than the
197                        # nested dict's leaf keys) would silently flatten
198                        # into the 2nd-level dict -- skip it visibly instead.
199                        print(
200                            "vault_config: config nesting deeper than 2 levels "
201                            f"is not supported; key '{key}' (line {lineno}) ignored",
202                            file=sys.stderr,
203                        )
204                        continue
205                    nested[key] = parse_scalar(value_str)
206                continue
207
208            if not value_str:
209                section[key] = {}
210                current_nested_key = key
211                current_nested_indent = indent
212                current_nested_leaf_indent = None
213            else:
214                section[key] = parse_scalar(value_str)
215                current_nested_key = None
216                current_nested_indent = 0
217                current_nested_leaf_indent = None
218        elif indent > 0:
219            # Indented line outside any section -- likely a typo
220            print(
221                f"vault_config: ignoring indented config line outside any section: {stripped!r}",
222                file=sys.stderr,
223            )
224
225    return result

233def _merge_config_dicts(
234    base: dict[str, Any], overlay: dict[str, Any]
235) -> dict[str, Any]:
236    """Deep-merge *overlay* over *base*, with *overlay* winning on conflict.
237
238    Dict values present in both are merged recursively key-by-key (matching
239    the parser's supported nesting depth, e.g. ``ai_models.codex``). A dict
240    value present only in *overlay* is added as a new section. Any other
241    value type -- including top-level scalars -- is replaced outright by the
242    *overlay* value.
243    """
244    merged = dict(base)
245    for key, overlay_value in overlay.items():
246        base_value = merged.get(key)
247        if isinstance(base_value, dict) and isinstance(overlay_value, dict):
248            merged[key] = _merge_config_dicts(base_value, overlay_value)
249        else:
250            merged[key] = overlay_value
251    return merged

254# Legacy config aliases for deployments that predate the parsight rename
255# (the backend product was formerly "par-mem"). ``_apply_legacy_aliases``
256# folds these into their canonical spellings at load time so every consumer
257# (get_config, load_typed_config, validate_config) sees canonical names.
258_LEGACY_SECTION_ALIASES: dict[str, str] = {"par_mem": "parsight"}
259_LEGACY_ENUM_ALIASES: dict[tuple[str, str], dict[str, str]] = {
260    ("search", "backend"): {"par-mem": "parsight"},
261}
262

264def _apply_legacy_aliases(config: dict[str, Any]) -> dict[str, Any]:
265    """Fold legacy section/enum spellings into their canonical names.
266
267    Compat for configs that predate the parsight rename (the backend product
268    was formerly "par-mem"): the legacy ``par_mem`` section is merged under
269    ``parsight`` (the canonical section wins per key when both are present),
270    and the legacy ``par-mem`` value of ``search.backend`` is rewritten to
271    ``parsight``. Applied to each file's parse *before* the
272    ``config.local.yaml`` merge so overlay precedence (local wins) is
273    preserved across the alias. Mutates and returns *config*.
274    """
275    for old_name, new_name in _LEGACY_SECTION_ALIASES.items():
276        old_section = config.get(old_name)
277        if not isinstance(old_section, dict):
278            continue
279        new_section = config.get(new_name)
280        if isinstance(new_section, dict):
281            merged = dict(old_section)
282            merged.update(new_section)  # canonical section wins per key
283            config[new_name] = merged
284        else:
285            config[new_name] = dict(old_section)
286        del config[old_name]
287    for (section, key), aliases in _LEGACY_ENUM_ALIASES.items():
288        section_dict = config.get(section)
289        if isinstance(section_dict, dict):
290            value = section_dict.get(key)
291            if isinstance(value, str) and value in aliases:
292                section_dict[key] = aliases[value]
293    return config

296@functools.lru_cache(maxsize=8)
297def _load_config_cached(vault: Path | None = None) -> dict[str, Any]:
298    """Read and merge the config files once per vault (QA-101 cache layer).
299
300    Returns the *shared* cached parsed dict. Callers must treat it as
301    read-only -- the public :func:`load_config` wrapper hands each caller
302    a deep copy of this value. The cache cap is 8 so alternating vaults
303    (multi-vault setups, tests with ``tmp_vault`` plus the real vault) stop
304    evicting each other on every call.
305
306    ``config.local.yaml`` is an optional gitignored overlay read from the
307    same vault directory. When present, it is deep-merged over ``config.yaml``
308    (section-by-section, with local values winning on conflict) so users can
309    keep secrets in the local-only file while git-syncing a secret-free
310    ``config.yaml``, or vice versa. Legacy section/enum spellings are
311    normalized per file before the merge (see :func:`_apply_legacy_aliases`).
312
313    Both files are read within the same cached call, so the cache covers
314    them jointly -- call :func:`clear_config_cache` to invalidate it (in
315    tests, or after either file changes).
316
317    Args:
318        vault: Optional vault path. Defaults to resolve_vault().
319
320    Returns an empty dict when both files are missing or unreadable.
321    """
322    if vault is None:
323        vault = resolve_vault()
324
325    config: dict[str, Any] = {}
326    config_path = vault / "config.yaml"
327    if config_path.is_file():
328        try:
329            content = config_path.read_text(encoding="utf-8")
330            config = _apply_legacy_aliases(_parse_config_yaml(content))
331        except (OSError, UnicodeDecodeError):
332            config = {}
333
334    local_path = vault / "config.local.yaml"
335    if local_path.is_file():
336        try:
337            local_content = local_path.read_text(encoding="utf-8")
338            local_config = _apply_legacy_aliases(_parse_config_yaml(local_content))
339            config = _merge_config_dicts(config, local_config)
340        except (OSError, UnicodeDecodeError):
341            pass
342
343    return config

346def config_key_sources(vault: Path | None = None) -> dict[tuple[str, str], str]:
347    """Map each first-level ``section.key`` to the config file it came from.
348
349    SEC-007: callers must distinguish values that originate in the
350    git-synced ``config.yaml`` from those in the machine-local
351    ``config.local.yaml`` before honoring them (e.g. network-affecting
352    ``anthropic_env`` keys). Values are ``"config.yaml"`` /
353    ``"config.local.yaml"``; keys absent from both files are absent from
354    the map. On conflict ``config.local.yaml`` wins, mirroring
355    :func:`load_config`'s merge order.
356
357    Deliberately uncached: it re-reads the two small files on each call so
358    it can never disagree with a :func:`clear_config_cache` call in tests.
359    """
360    if vault is None:
361        vault = resolve_vault()
362
363    sources: dict[tuple[str, str], str] = {}
364    for file_name in ("config.yaml", "config.local.yaml"):
365        path = vault / file_name
366        if not path.is_file():
367            continue
368        try:
369            data = _parse_config_yaml(path.read_text(encoding="utf-8"))
370        except (OSError, UnicodeDecodeError):
371            continue
372        for section, values in data.items():
373            if not isinstance(values, dict):
374                continue
375            for key in values:
376                sources[(str(section), str(key))] = file_name
377    return sources

380def _deep_copy_config(obj: Any) -> Any:
381    """Return a recursive copy of a config-derived value.
382
383    Mirrors the shape ``_parse_config_yaml`` can produce: dicts at up to two
384    nested levels (section -> nested_section -> scalar), lists only inside
385    ``related``/``tags`` frontmatter (not in config), and scalars (str/int/
386    float/bool/None). Handled without ``copy.deepcopy`` so the cost is
387    proportional to the (small) config tree, not a generic Python object walk.
388    """
389    if isinstance(obj, dict):
390        return {k: _deep_copy_config(v) for k, v in obj.items()}
391    if isinstance(obj, list):
392        return [_deep_copy_config(v) for v in obj]
393    return obj

403def load_config(vault: Path | None = None) -> dict[str, Any]:
404    """Load ``config.yaml`` from the vault, layered with ``config.local.yaml``.
405
406    Public wrapper over :func:`_load_config_cached`: the file read + parse is
407    cached per-process (keyed on the vault path), and every call returns a
408    fresh deep copy of the cached dict.
409
410    ARC-034 / QA-101: a caller that mutates the returned dict (e.g.
411    ``config["summarizer"]["model"] = "claude-..."``) cannot corrupt the
412    values seen by later callers -- the deep copy is made per call, outside
413    the cache. (Previously the copy happened inside the cached function, so
414    ``functools.lru_cache`` handed every caller the same copy object and the
415    documented isolation did not hold.)
416
417    Args:
418        vault: Optional vault path. Defaults to resolve_vault().
419
420    Returns an empty dict when both files are missing or unreadable.
421    """
422    return _deep_copy_config(_load_config_cached(vault))

440def clamp_timeout(
441    value: int | float | None,
442    default: int | float,
443    lo: int | float = 1,
444    hi: int | float = 3600,
445) -> int | float:
446    """Coerce a configured timeout into ``[lo, hi]`` (SEC-024).
447
448    Config-sourced timeouts previously reached ``subprocess.run(timeout=...)``
449    unvalidated: ``float('nan')`` and negatives raise at call time, and
450    ``inf`` silently means "no timeout at all". Non-finite, non-positive, or
451    non-numeric input (including ``True``/``False``, which ``isinstance``
452    would otherwise treat as ``int``) falls back to *default*.
453
454    Args:
455        value: Raw configured value.
456        default: Fallback for invalid input.
457        lo: Lower bound (inclusive) for valid values.
458        hi: Upper bound (inclusive) for valid values.
459
460    Returns:
461        The clamped timeout as int or float.
462    """
463    if isinstance(value, bool) or not isinstance(value, (int, float)):
464        return default
465    if not math.isfinite(float(value)) or value <= 0:
466        return default
467    return min(max(value, lo), hi)

470def get_config(
471    section: str, key: str, default: Any = None, vault: Path | None = None
472) -> Any:
473    """Look up a config value with fallback to *default*.
474
475    ARC-007 adapter over the typed schema. Resolution order:
476
477    1. Key present in the parsed config dict (config.yaml layered with
478       config.local.yaml) → that value, unchanged. An explicit ``null`` thus
479       returns ``None``, so users can disable optional features with e.g.
480       ``ai_model: null``.
481    2. Key absent but the schema declares a real default for it
482       (:data:`vault_schema` field default) → the schema default.
483    3. Otherwise → *default*.
484
485    Args:
486        section: Top-level section name (e.g. ``"session_start_hook"``).
487        key: Key within the section (e.g. ``"max_chars"``).
488        default: Value returned when the key is absent and the schema
489            declares no default for it.
490        vault: Optional vault path so multi-vault callers read the config of
491            the vault they are operating on (ARC-101). Defaults to
492            resolve_vault() — hook-path callers must pass the vault resolved
493            from the hook payload instead of relying on the process cwd.
494
495    Returns:
496        The configured value (which may be ``None`` if explicitly set), the
497        schema default, or *default*.
498    """
499    config = load_config(vault=vault)
500    section_dict = config.get(section)
501    if isinstance(section_dict, dict) and key in section_dict:
502        return section_dict[key]
503
504    cfg = load_typed_config(vault=vault)
505    section_obj = getattr(cfg, section, None)
506    if section_obj is not None:
507        for field in dataclasses.fields(section_obj):
508            if field.name != key:
509                continue
510            if field.default is not dataclasses.MISSING and field.default is not None:
511                return field.default
512            if field.default_factory is not dataclasses.MISSING:  # pragma: no cover
513                return field.default_factory()
514            break
515    return default

518def _load_typed_config_cached(vault: Path | None = None) -> VaultAppConfig:
519    """Build the typed config from the cached parsed dict (compat name).
520
521    ARC-007: the separate lru cache was removed. It could serve stale values
522    after a caller cleared only the parse cache (the common test/invalidation
523    idiom), and it only saved the cheap dict-to-dataclass mapping -- the
524    expensive work (file read + YAML parse) is cached inside
525    :func:`_load_config_cached`. Kept as an alias because the name is
526    re-exported.
527
528    QA-101: builds directly from the shared cached dict without a deep copy.
529    This is safe because ``VaultAppConfig.from_dict`` assigns only the
530    parsed dict's scalar leaf values onto the section dataclasses (the
531    config parser produces no lists), so the dataclass tree never aliases
532    a mutable leaf of the cached dict.
533    """
534    return VaultAppConfig.from_dict(_load_config_cached(vault))

537def load_typed_config(vault: Path | None = None) -> VaultAppConfig:

589def resolve_decay_params(vault: Path | None = None) -> tuple[float, float]:

615def apply_decay_score(
616    score: float,
617    mtime: float,
618    now: float,
619    *,
620    half_life_days: float | None = None,
621    min_factor: float | None = None,
622    vault: Path | None = None,
623) -> float:
624    """Apply exponential temporal decay to an embedding/RRF search score.
625
626    The decay factor is ``min_factor + (1 - min_factor) * e^(-lambda *
627    age_days)`` where ``lambda = ln(2) / half_life_days`` — so a note exactly
628    ``half_life_days`` old decays to the midpoint between 1.0 and
629    ``min_factor``.
630
631    PRF-103: batch callers pass *half_life_days* and *min_factor* explicitly
632    (resolved once per search via :func:`resolve_decay_params`) so scored rows
633    never trigger config reads. When omitted, both are read from config
634    (``embeddings.decay_half_life_days``, default 90, and
635    ``embeddings.decay_min_factor``, default 0.5) — the pre-PRF-103 behavior
636    for non-hot callers.
637
638    Args:
639        score: Raw similarity / RRF score.
640        mtime: Note file modification time (Unix timestamp).
641        now: Reference timestamp (typically ``time.time()``); pass 0.0 to skip
642            decay (used when the caller has already decided decay is disabled).
643        half_life_days: Decay half-life in days; None reads it from config.
644        min_factor: Floor multiplier for very old notes; None reads it from
645            config.
646        vault: Optional vault path used only when reading config because the
647            decay parameters were omitted.
648
649    Returns:
650        Decay-adjusted score. When ``mtime`` is 0/missing the score is returned
651        unchanged (the original code path that gated on ``if mtime``).
652    """
653    if half_life_days is None or min_factor is None:
654        half_life_days, min_factor = resolve_decay_params(vault)
655    if not mtime:
656        return score
657    age_days = max(0.0, (now - mtime) / 86400.0)
658    lam = math.log(2) / half_life_days
659    decay = min_factor + (1.0 - min_factor) * math.exp(-lam * age_days)
660    return score * decay

161

164    quotes, and the split must not let one toggle the quote state and
