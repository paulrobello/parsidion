 49def _split_list_items(text: str) -> list[str]:
 50    """Split a comma-separated list, respecting quoted strings."""
 51    items: list[str] = []
 52    current: list[str] = []
 53    in_quote: str | None = None
 54
 55    for ch in text:
 56        if in_quote:
 57            current.append(ch)
 58            if ch == in_quote:
 59                in_quote = None
 60        elif ch in ('"', "'"):
 61            in_quote = ch
 62            current.append(ch)
 63        elif ch == ",":
 64            items.append("".join(current).strip())
 65            current = []
 66        else:
 67            current.append(ch)
 68
 69    remaining = "".join(current).strip()
 70    if remaining:
 71        items.append(remaining)
 72
 73    return items

 76def _parse_scalar(value: str) -> Any:
 77    """Parse a scalar YAML value into a Python type.
 78
 79    Handles booleans, None/null, integers, floats, quoted strings, and bare
 80    strings. Date strings (YYYY-MM-DD) are kept as strings for simplicity.
 81    """
 82    # Strip surrounding quotes
 83    if len(value) >= 2 and value[0] == value[-1] and value[0] in ('"', "'"):
 84        return value[1:-1]
 85
 86    lower = value.lower()
 87    if lower in ("true", "yes"):
 88        return True
 89    if lower in ("false", "no"):
 90        return False
 91    if lower in ("null", "~", ""):
 92        return None
 93
 94    # Try integer
 95    try:
 96        return int(value)
 97    except ValueError:
 98        pass
 99
100    # Try float
101    try:
102        return float(value)
103    except ValueError:
104        pass
105
106    return value

109def _parse_list_item(value: str) -> str:
110    """Parse a YAML list item, keeping it as a string.
111
112    Unlike ``_parse_scalar``, list items are never coerced to bool/int/float:
113    frontmatter list fields (``tags``, ``sources``, ``related``) are always
114    string-valued, and coercing e.g. ``tags: [2026, python]`` to an int makes
115    the tag silently unfindable downstream. Surrounding quotes are stripped.
116    """
117    if len(value) >= 2 and value[0] == value[-1] and value[0] in ('"', "'"):
118        return value[1:-1]
119    return value

122def _strip_inline_comment(value: str) -> str:
123    """Strip a trailing ``# comment`` from a YAML value, respecting quotes."""
124    in_quote: str | None = None
125    for i, ch in enumerate(value):
126        if in_quote:
127            if ch == in_quote:
128                in_quote = None
129        elif ch in ('"', "'"):
130            in_quote = ch
131        elif ch == "#" and i > 0 and value[i - 1] in (" ", "\t"):
132            return value[:i].rstrip()
133    return value

141def _parse_config_yaml(text: str) -> dict[str, Any]:
142    """Parse a simple YAML config with limited nesting.
143
144    Handles top-level scalars, section dicts, and one additional mapping level
145    for config sections such as ``ai_models.codex.small``::
146
147        top_key: value
148        section:
149          nested_key: value
150          nested_section:
151            leaf_key: value
152    """
153    result: dict[str, Any] = {}
154    current_section: str | None = None
155    current_nested_key: str | None = None
156    current_nested_indent = 0
157    # Indent of leaf keys inside the current nested dict; anything indented
158    # deeper is a 3rd nesting level, which this parser does not support.
159    current_nested_leaf_indent: int | None = None
160
161    for lineno, line in enumerate(text.splitlines(), 1):
162        stripped = line.strip()
163        if not stripped or stripped.startswith("#"):
164            continue
165
166        indent = len(line) - len(line.lstrip())
167        colon_idx = stripped.find(":")
168        if colon_idx == -1:
169            print(
170                f"vault_config: ignoring unparsable config line: {stripped!r}",
171                file=sys.stderr,
172            )
173            continue
174
175        key = stripped[:colon_idx].strip()
176        value_str = stripped[colon_idx + 1 :].strip()
177
178        if not key:
179            print(
180                f"vault_config: ignoring config line with empty key: {stripped!r}",
181                file=sys.stderr,
182            )
183            continue
184
185        if indent == 0:
186            if not value_str:
187                # Section header -- start collecting nested keys
188                current_section = key
189                current_nested_key = None
190                current_nested_indent = 0
191                current_nested_leaf_indent = None
192                result[key] = {}
193            else:
194                value_str = _strip_inline_comment(value_str)
195                result[key] = _parse_scalar(value_str)
196                current_section = None
197                current_nested_key = None
198                current_nested_indent = 0
199                current_nested_leaf_indent = None
200        elif current_section is not None and indent > 0:
201            value_str = _strip_inline_comment(value_str)
202            section = result.get(current_section)
203            if not isinstance(section, dict):
204                continue
205
206            if (
207                current_nested_key is not None
208                and indent > current_nested_indent
209                and isinstance(section.get(current_nested_key), dict)
210            ):
211                nested = section[current_nested_key]
212                if isinstance(nested, dict):
213                    if current_nested_leaf_indent is None:
214                        current_nested_leaf_indent = indent
215                    if not value_str or indent > current_nested_leaf_indent:
216                        # A 3rd nesting level (a sub-section header inside a
217                        # nested dict, or a key indented deeper than the
218                        # nested dict's leaf keys) would silently flatten
219                        # into the 2nd-level dict -- skip it visibly instead.
220                        print(
221                            "vault_config: config nesting deeper than 2 levels "
222                            f"is not supported; key '{key}' (line {lineno}) ignored",
223                            file=sys.stderr,
224                        )
225                        continue
226                    nested[key] = _parse_scalar(value_str)
227                continue
228
229            if not value_str:
230                section[key] = {}
231                current_nested_key = key
232                current_nested_indent = indent
233                current_nested_leaf_indent = None
234            else:
235                section[key] = _parse_scalar(value_str)
236                current_nested_key = None
237                current_nested_indent = 0
238                current_nested_leaf_indent = None
239        elif indent > 0:
240            # Indented line outside any section -- likely a typo
241            print(
242                f"vault_config: ignoring indented config line outside any section: {stripped!r}",
243                file=sys.stderr,
244            )
245
246    return result

254def _merge_config_dicts(
255    base: dict[str, Any], overlay: dict[str, Any]
256) -> dict[str, Any]:
257    """Deep-merge *overlay* over *base*, with *overlay* winning on conflict.
258
259    Dict values present in both are merged recursively key-by-key (matching
260    the parser's supported nesting depth, e.g. ``ai_models.codex``). A dict
261    value present only in *overlay* is added as a new section. Any other
262    value type -- including top-level scalars -- is replaced outright by the
263    *overlay* value.
264    """
265    merged = dict(base)
266    for key, overlay_value in overlay.items():
267        base_value = merged.get(key)
268        if isinstance(base_value, dict) and isinstance(overlay_value, dict):
269            merged[key] = _merge_config_dicts(base_value, overlay_value)
270        else:
271            merged[key] = overlay_value
272    return merged

275@functools.lru_cache(maxsize=8)
276def load_config(vault: Path | None = None) -> dict[str, Any]:
277    """Load ``config.yaml`` from the vault, layered with ``config.local.yaml``.
278
279    ``config.local.yaml`` is an optional gitignored overlay read from the
280    same vault directory. When present, it is deep-merged over ``config.yaml``
281    (section-by-section, with local values winning on conflict) so users can
282    keep secrets in the local-only file while git-syncing a secret-free
283    ``config.yaml``, or vice versa.
284
285    Results are cached per-process via ``functools.lru_cache``. Both files
286    are read within the same cached call, so the cache covers them jointly --
287    call ``load_config.cache_clear()`` to invalidate the cache in tests (or
288    after either file changes) when the vault path has been changed.
289
290    ARC-034: returns a deep copy so a caller that mutates the returned dict
291    (e.g. ``config["summarizer"]["model"] = "claude-..."``) cannot corrupt the
292    cached values for the rest of the process. The cache cap was also raised
293    from 1 to 8 so alternating vaults (multi-vault setups, tests with
294    ``tmp_vault`` plus the real vault) stop evicting each other on every call.
295
296    Args:
297        vault: Optional vault path. Defaults to resolve_vault().
298
299    Returns an empty dict when both files are missing or unreadable.
300    """
301    if vault is None:
302        vault = resolve_vault()
303
304    config: dict[str, Any] = {}
305    config_path = vault / "config.yaml"
306    if config_path.is_file():
307        try:
308            content = config_path.read_text(encoding="utf-8")
309            config = _parse_config_yaml(content)
310        except (OSError, UnicodeDecodeError):
311            config = {}
312
313    local_path = vault / "config.local.yaml"
314    if local_path.is_file():
315        try:
316            local_content = local_path.read_text(encoding="utf-8")
317            local_config = _parse_config_yaml(local_content)
318            config = _merge_config_dicts(config, local_config)
319        except (OSError, UnicodeDecodeError):
320            pass
321
322    return _deep_copy_config(config)

325def _deep_copy_config(obj: Any) -> Any:
326    """Return a recursive copy of a config-derived value.
327
328    Mirrors the shape ``_parse_config_yaml`` can produce: dicts at up to two
329    nested levels (section -> nested_section -> scalar), lists only inside
330    ``related``/``tags`` frontmatter (not in config), and scalars (str/int/
331    float/bool/None). Handled without ``copy.deepcopy`` so the cost is
332    proportional to the (small) config tree, not a generic Python object walk.
333    """
334    if isinstance(obj, dict):
335        return {k: _deep_copy_config(v) for k, v in obj.items()}
336    if isinstance(obj, list):
337        return [_deep_copy_config(v) for v in obj]
338    return obj

346def get_config(section: str, key: str, default: Any = None) -> Any:
347    """Look up a config value with fallback to *default*.
348
349    Distinguishes between a key that is absent (returns *default*) and a key
350    that is explicitly set to ``null`` in config.yaml (returns ``None``).  This
351    allows users to disable optional features by setting e.g. ``ai_model: null``.
352
353    Args:
354        section: Top-level section name (e.g. ``"session_start_hook"``).
355        key: Key within the section (e.g. ``"max_chars"``).
356        default: Value returned when the key is absent from the config file.
357
358    Returns:
359        The configured value (which may be ``None`` if explicitly set), or
360        *default* when the key is absent.
361    """
362    config = load_config()
363    section_dict = config.get(section)
364    if isinstance(section_dict, dict):
365        if key in section_dict:
366            return section_dict[key]
367    return default

382def apply_decay_score(score: float, mtime: float, now: float) -> float:
383    """Apply exponential temporal decay to an embedding/RRF search score.
384
385    Reads ``embeddings.decay_half_life_days`` (default 90) and
386    ``embeddings.decay_min_factor`` (default 0.5) from config. The decay factor
387    is ``min_factor + (1 - min_factor) * e^(-lambda * age_days)`` where
388    ``lambda = ln(2) / half_life_days`` — so a note exactly ``half_life_days``
389    old decays to the midpoint between 1.0 and ``min_factor``.
390
391    Args:
392        score: Raw similarity / RRF score.
393        mtime: Note file modification time (Unix timestamp).
394        now: Reference timestamp (typically ``time.time()``); pass 0.0 to skip
395            decay (used when the caller has already decided decay is disabled).
396
397    Returns:
398        Decay-adjusted score. When ``mtime`` is 0/missing the score is returned
399        unchanged (the original code path that gated on ``if mtime``).
400    """
401    half_life: float = get_config("embeddings", "decay_half_life_days", 90.0)
402    min_factor: float = get_config("embeddings", "decay_min_factor", 0.5)
403    if not mtime:
404        return score
405    age_days = max(0.0, (now - mtime) / 86400.0)
406    lam = math.log(2) / half_life
407    decay = min_factor + (1.0 - min_factor) * math.exp(-lam * age_days)
408    return score * decay

538def validate_config() -> list[str]:
539    """Validate config.yaml against the known schema.
540
541    Checks for unknown sections, unknown keys within known sections, and
542    type mismatches. Warnings are informational -- never raises.
543
544    Returns:
545        A list of warning strings (empty when config is valid or absent).
546    """
547    config = load_config()
548    if not config:
549        return []
550
551    warnings: list[str] = []
552    known_sections = set(_CONFIG_SCHEMA.keys())
553
554    for section, section_value in config.items():
555        if section not in known_sections:
556            warnings.append(f"config.yaml: unknown section '{section}'")
557            continue
558
559        if not isinstance(section_value, dict):
560            warnings.append(
561                f"config.yaml: section '{section}' should be a mapping, got {type(section_value).__name__}"
562            )
563            continue
564
565        schema_keys = _CONFIG_SCHEMA[section]
566        for key, value in section_value.items():
567            if key not in schema_keys:
568                warnings.append(f"config.yaml: unknown key '{section}.{key}'")
569                continue
570            expected_types = schema_keys[key]
571            if value is not None and not isinstance(value, expected_types):
572                type_names = " | ".join(
573                    t.__name__ for t in expected_types if t is not type(None)
574                )
575                warnings.append(
576                    f"config.yaml: '{section}.{key}' expected {type_names}, "
577                    f"got {type(value).__name__}"
578                )
579
580    return warnings

50def _split_list_items(text: str) -> list[str]:
51    """Split a comma-separated list, respecting quoted strings."""
52    items: list[str] = []
53    current: list[str] = []
54    in_quote: str | None = None
55
56    for ch in text:
57        if in_quote:
58            current.append(ch)
59            if ch == in_quote:
60                in_quote = None
61        elif ch in ('"', "'"):
62            in_quote = ch
63            current.append(ch)
64        elif ch == ",":
65            items.append("".join(current).strip())
66            current = []
67        else:
68            current.append(ch)
69
70    remaining = "".join(current).strip()
71    if remaining:
72        items.append(remaining)

def _parse_config_yaml(text: str) -> dict[str, typing.Any]: 
142def _parse_config_yaml(text: str) -> dict[str, Any]:
143    """Parse a simple YAML config with limited nesting.
144
145    Handles top-level scalars, section dicts, and one additional mapping level
146    for config sections such as ``ai_models.codex.small``::
147
148        top_key: value
149        section:
150          nested_key: value
151          nested_section:
152            leaf_key: value
153    """
154    result: dict[str, Any] = {}
155    current_section: str | None = None
156    current_nested_key: str | None = None
157    current_nested_indent = 0
158    # Indent of leaf keys inside the current nested dict; anything indented
159    # deeper is a 3rd nesting level, which this parser does not support.
160    current_nested_leaf_indent: int | None = None
161
162    for lineno, line in enumerate(text.splitlines(), 1):
163        stripped = line.strip()
164        if not stripped or stripped.startswith("#"):
165            continue
166
167        indent = len(line) - len(line.lstrip())
168        colon_idx = stripped.find(":")
169        if colon_idx == -1:
170            print(
171                f"vault_config: ignoring unparsable config line: {stripped!r}",
172                file=sys.stderr,
173            )
174            continue
175
176        key = stripped[:colon_idx].strip()
177        value_str = stripped[colon_idx + 1 :].strip()
178
179        if not key:
180            print(
181                f"vault_config: ignoring config line with empty key: {stripped!r}",
182                file=sys.stderr,
183            )
184            continue
185
186        if indent == 0:
187            if not value_str:
188                # Section header -- start collecting nested keys
189                current_section = key
190                current_nested_key = None
191                current_nested_indent = 0
192                current_nested_leaf_indent = None
193                result[key] = {}
194            else:
195                value_str = _strip_inline_comment(value_str)
196                result[key] = _parse_scalar(value_str)
197                current_section = None
198                current_nested_key = None
199                current_nested_indent = 0
200                current_nested_leaf_indent = None
201        elif current_section is not None and indent > 0:
202            value_str = _strip_inline_comment(value_str)
203            section = result.get(current_section)
204            if not isinstance(section, dict):
205                continue
206
207            if (
208                current_nested_key is not None
209                and indent > current_nested_indent
210                and isinstance(section.get(current_nested_key), dict)
211            ):
212                nested = section[current_nested_key]
213                if isinstance(nested, dict):
214                    if current_nested_leaf_indent is None:
215                        current_nested_leaf_indent = indent
216                    if not value_str or indent > current_nested_leaf_indent:
217                        # A 3rd nesting level (a sub-section header inside a
218                        # nested dict, or a key indented deeper than the
219                        # nested dict's leaf keys) would silently flatten
220                        # into the 2nd-level dict -- skip it visibly instead.
221                        print(
222                            "vault_config: config nesting deeper than 2 levels "
223                            f"is not supported; key '{key}' (line {lineno}) ignored",
224                            file=sys.stderr,
225                        )
226                        continue
227                    nested[key] = _parse_scalar(value_str)
228                continue
229
230            if not value_str:
231                section[key] = {}
232                current_nested_key = key
233                current_nested_indent = indent
234                current_nested_leaf_indent = None
235            else:
236                section[key] = _parse_scalar(value_str)
237                current_nested_key = None
238                current_nested_indent = 0
239                current_nested_leaf_indent = None
240        elif indent > 0:
241            # Indented line outside any section -- likely a typo
242            print(
243                f"vault_config: ignoring indented config line outside any section: {stripped!r}",
244                file=sys.stderr,
245            )
246
247    return result

def _merge_config_dicts( base: dict[str, typing.Any], overlay: dict[str, typing.Any]) -> dict[str, typing.Any]: 
255def _merge_config_dicts(
256    base: dict[str, Any], overlay: dict[str, Any]
257) -> dict[str, Any]:
258    """Deep-merge *overlay* over *base*, with *overlay* winning on conflict.
259
260    Dict values present in both are merged recursively key-by-key (matching
261    the parser's supported nesting depth, e.g. ``ai_models.codex``). A dict
262    value present only in *overlay* is added as a new section. Any other
263    value type -- including top-level scalars -- is replaced outright by the
264    *overlay* value.
265    """
266    merged = dict(base)
267    for key, overlay_value in overlay.items():
268        base_value = merged.get(key)
269        if isinstance(base_value, dict) and isinstance(overlay_value, dict):
270            merged[key] = _merge_config_dicts(base_value, overlay_value)
271        else:
272            merged[key] = overlay_value
273    return merged

def apply_decay_score(score: float, mtime: float, now: float) -> float: 
383def apply_decay_score(score: float, mtime: float, now: float) -> float:
384    """Apply exponential temporal decay to an embedding/RRF search score.
385
386    Reads ``embeddings.decay_half_life_days`` (default 90) and
387    ``embeddings.decay_min_factor`` (default 0.5) from config. The decay factor
388    is ``min_factor + (1 - min_factor) * e^(-lambda * age_days)`` where
389    ``lambda = ln(2) / half_life_days`` — so a note exactly ``half_life_days``
390    old decays to the midpoint between 1.0 and ``min_factor``.
391
392    Args:
393        score: Raw similarity / RRF score.
394        mtime: Note file modification time (Unix timestamp).
395        now: Reference timestamp (typically ``time.time()``); pass 0.0 to skip
396            decay (used when the caller has already decided decay is disabled).
397
398    Returns:
399        Decay-adjusted score. When ``mtime`` is 0/missing the score is returned
400        unchanged (the original code path that gated on ``if mtime``).
401    """
402    half_life: float = get_config("embeddings", "decay_half_life_days", 90.0)
403    min_factor: float = get_config("embeddings", "decay_min_factor", 0.5)
404    if not mtime:
405        return score
406    age_days = max(0.0, (now - mtime) / 86400.0)
407    lam = math.log(2) / half_life
408    decay = min_factor + (1.0 - min_factor) * math.exp(-lam * age_days)
409    return score * decay
