core.vault_health

vault_health — composite vault health score (ENH-007).

Seven scored dimensions (0–100 each) combined into a weighted overall grade. Each dimension returns a DimensionScore carrying its own action string — a score with no concrete next action is just a number, so dimensions that cannot name a command deliberately leave action as None when healthy.

This module is stdlib-only (it is imported by vault_stats, the MCP server, and indirectly by hooks). The display layer (vault_stats) may freely use rich; this scoring layer must not. The constraint is enforced structurally by tests/test_stdlib_only.py.

Reuse contract: frontmatter validity and broken-wikilink counts come from vault_doctor.scan_notes_readonly — the same check_note path the repair pipeline uses. There is no second validator here.

All score_* functions degrade rather than raise: a missing graph.json, a missing embeddings.db, or an empty vault produces a low score with a detail string, never a crash.

 62# ---------------------------------------------------------------------------
 63# Dimension weights — single source of truth (Step 2 of the plan).
 64# ---------------------------------------------------------------------------
 65# Rationale: the read path (index + embeddings + queue) weighs heaviest
 66# because staleness or a dead-letter backlog silently breaks every other
 67# feature. Metadata + connectivity weigh next — they bound search and graph
 68# quality. Tag/file hygiene are real but lower-impact operationally.
 69# Adjustments are cheap; the rationale matters more than the numbers.
 70DIMENSION_WEIGHTS: dict[str, int] = {
 71    "index_freshness": 20,
 72    "queue_health": 20,
 73    "graph_connectivity": 15,
 74    "metadata_quality": 15,
 75    "embedding_coverage": 10,
 76    "tag_hygiene": 10,
 77    "file_hygiene": 10,
 78}

 82    (90, "A"),

 94@dataclass(frozen=True)
 95class DimensionScore:
 96    """One scored dimension.
 97
 98    Attributes:
 99        name: Canonical dimension key (matches a ``DIMENSION_WEIGHTS`` entry).
100        score: 0–100 integer.
101        weight: Weight (matches ``DIMENSION_WEIGHTS[name]``); denormalised
102            onto each score so a serialized report is self-describing.
103        detail: Human-readable finding (what the score observed).
104        action: Concrete command to run when unhealthy, or ``None`` when the
105            dimension is healthy. The plan mandates: a dimension that cannot
106            name an action should not score; here that maps to ``action=None``
107            rather than withholding the score.
108    """
109
110    name: str
111    score: int
112    weight: int
113    detail: str
114    action: str | None = None

146def _age_score(age_days: float, *, fresh: float = 1.0, stale: float = 30.0) -> int:
147    """Linear 100→0 between ``fresh`` and ``stale`` days.
148
149    100 at ``age ≤ fresh``, 0 at ``age ≥ stale``, clamped to [0, 100].
150    """
151    if age_days <= fresh:
152        return 100
153    if age_days >= stale:
154        return 0
155    span = stale - fresh
156    return max(0, min(100, int(round(100 * (1 - (age_days - fresh) / span)))))

159def _grade_for(score: int) -> str:
160    """Return the letter grade for an overall score."""
161    for threshold, letter in _GRADE_BANDS:
162        if score >= threshold:
163            return letter
164    return "F"

172def _graph_meta(vault: Path) -> dict | None:
173    """Return the parsed ``graph.json`` ``meta`` block, or None if absent."""
174    graph_path = vault / "graph.json"
175    try:
176        raw = graph_path.read_text(encoding="utf-8")
177    except OSError:
178        return None
179    try:
180        return json.loads(raw).get("meta") or None
181    except json.JSONDecodeError:
182        return None

185def _parse_iso_z(ts: str) -> float | None:
186    """Parse an ISO-8601 ``Z`` timestamp into epoch seconds, or None."""
187    if not ts:
188        return None
189    # Accept "...Z"; fromisoformat handles offsets in 3.11+ but not bare Z
190    # until 3.11 — normalise first.
191    cleaned = ts.strip()
192    if cleaned.endswith("Z"):
193        cleaned = cleaned[:-1] + "+00:00"
194    try:
195        from datetime import datetime
196
197        return datetime.fromisoformat(cleaned).timestamp()
198    except ValueError:
199        return None

202def score_index_freshness(vault: Path) -> DimensionScore:
203    """Index freshness: graph.json age + note_index age, averaged."""
204    weight = DIMENSION_WEIGHTS["index_freshness"]
205    graph_meta = _graph_meta(vault)
206    now = time.time()
207
208    # --- graph.json component ---
209    if graph_meta is None:
210        graph_score = 0
211        graph_detail = "graph.json missing or unreadable"
212        graph_age_days: float | None = None
213    else:
214        generated = graph_meta.get("generated", "")
215        gen_ts = _parse_iso_z(str(generated))
216        if gen_ts is None:
217            graph_score = 0
218            graph_detail = "graph.json meta.generated is unparseable"
219            graph_age_days = None
220        else:
221            graph_age_days = max(0.0, (now - gen_ts) / 86400)
222            graph_score = _age_score(graph_age_days)
223            graph_detail = f"graph.json is {int(graph_age_days)} day(s) old"
224
225    # --- note_index component ---
226    try:
227        index_age_s = note_index_age(vault)
228    except Exception:  # noqa: BLE001 — note_index_age reads files; never raise
229        index_age_s = 0.0
230    # Treat as days of staleness, capped at the curve ceiling so a vault that
231    # has never been indexed scores 0 rather than 100 (age==0 would be fresh).
232    if index_age_s <= 0:
233        index_score = 100
234        index_detail = "note_index is current"
235    else:
236        index_age_days = index_age_s / 86400
237        index_score = _age_score(index_age_days)
238        index_detail = f"note_index is {int(index_age_days)} day(s) behind disk"
239
240    overall = (graph_score + index_score) // 2
241    detail = (
242        f"{graph_detail}; {index_detail}" if graph_meta is not None else graph_detail
243    )
244    action = (
245        "uv run install.py --schedule-summarizer --rebuild-graph"
246        if overall < 90
247        else None
248    )
249    return DimensionScore(
250        name="index_freshness",
251        score=overall,
252        weight=weight,
253        detail=detail,
254        action=action,
255    )

258def score_queue_health(vault: Path) -> DimensionScore:
259    """Queue health: ``100 - pending*2 - dead_letters*5``, floored at 0.
260
261    Dead letters weigh more because they represent *permanently* lost
262    summaries (repeated failures), not merely delayed ones.
263    """
264    weight = DIMENSION_WEIGHTS["queue_health"]
265    pending = vault_metrics.collect_pending(vault)
266    dead = vault_metrics.collect_dead_letters(vault)
267    pending_n = pending.get("total", 0) if pending.get("exists") else 0
268    dead_n = dead.get("total", 0) if dead.get("exists") else 0
269
270    raw = 100 - (pending_n * 2) - (dead_n * 5)
271    score = max(0, min(100, raw))
272
273    bits: list[str] = []
274    if pending_n:
275        bits.append(f"{pending_n} pending")
276    if dead_n:
277        bits.append(f"{dead_n} dead-lettered")
278    if not bits:
279        detail = "queue is empty"
280        action: str | None = None
281    else:
282        detail = ", ".join(bits)
283        # Prefer dead-letter remediation when present — that's the silent loss.
284        if dead_n:
285            action = "vault-review     # inspect dead letters, then re-queue or clear"
286        else:
287            action = "env -u CLAUDECODE uv run --no-project ~/.claude/skills/parsidion/scripts/summarize_sessions.py"
288    return DimensionScore(
289        name="queue_health",
290        score=score,
291        weight=weight,
292        detail=detail,
293        action=action,
294    )

297def score_graph_connectivity(vault: Path) -> DimensionScore:
298    """Graph connectivity: ``100 × (1 - orphans / total)`` plus a cluster penalty.
299
300    Reuses ``vault_metrics.collect_graph`` so the definition of "orphan"
301    (no ``related`` AND no incoming links) matches what ``--graph`` reports.
302    """
303    weight = DIMENSION_WEIGHTS["graph_connectivity"]
304    conn = vault_metrics.open_db(vault)
305    if conn is None:
306        # No index → we can't measure connectivity. Score 0 with a clear detail
307        # rather than guessing from a filesystem walk (which has no link data).
308        return DimensionScore(
309            name="graph_connectivity",
310            score=0,
311            weight=weight,
312            detail="note_index DB absent — run update_index.py",
313            action="uv run --no-project ~/.claude/skills/parsidion/scripts/update_index.py",
314        )
315    try:
316        data = vault_metrics.collect_graph(conn)
317    finally:
318        conn.close()
319
320    total = data.get("total", 0)
321    if total == 0:
322        return DimensionScore(
323            name="graph_connectivity",
324            score=0,
325            weight=weight,
326            detail="vault is empty (no indexed notes)",
327            action=None,
328        )
329    orphans = len(data.get("isolated_notes", []))
330    orphan_ratio = orphans / total
331    # Isolated-cluster penalty: dangling_targets/total_targets indicates how
332    # many wikilinks point at nothing. Cap the penalty so an otherwise-good
333    # graph isn't dragged below 50 by a few stale links.
334    total_targets = data.get("total_targets", 0)
335    dangling = data.get("dangling_targets", 0)
336    dangling_ratio = (dangling / total_targets) if total_targets else 0.0
337    penalty = min(dangling_ratio * 25, 20)
338    raw = 100 * (1 - orphan_ratio) - penalty
339    score = max(0, min(100, int(round(raw))))
340
341    bits: list[str] = []
342    if orphans:
343        bits.append(f"{orphans} orphan note(s)")
344    if dangling:
345        bits.append(f"{dangling} dangling wikilink(s)")
346    detail = "; ".join(bits) if bits else "well-connected"
347    action = "vault-doctor --fix-all" if orphans or dangling else None
348    return DimensionScore(
349        name="graph_connectivity",
350        score=score,
351        weight=weight,
352        detail=detail,
353        action=action,
354    )

357def score_metadata_quality(vault: Path, *, scan=None) -> DimensionScore:
358    """Metadata quality: ``100 × (1 - notes_with_issues / total)``.
359
360    Reuses ``vault_doctor.scan_notes_readonly`` — the *same* ``check_note``
361    path the repair pipeline uses — so there is exactly one validator.
362
363    The optional ``scan`` kwarg lets tests inject a precomputed
364    ``ScanSummary`` to avoid re-walking a large vault per dimension.
365    """
366    weight = DIMENSION_WEIGHTS["metadata_quality"]
367    if scan is None:
368        try:
369            from doctor.scan import scan_notes_readonly
370
371            scan = scan_notes_readonly(vault)
372        except Exception:  # noqa: BLE001 — scan must not crash the report
373            return DimensionScore(
374                name="metadata_quality",
375                score=0,
376                weight=weight,
377                detail="metadata scan failed (see vault-doctor --dry-run)",
378                action="uv run --no-project ~/.claude/skills/parsidion/scripts/vault_doctor.py --dry-run",
379            )
380    total = scan.total_notes
381    if total == 0:
382        return DimensionScore(
383            name="metadata_quality",
384            score=0,
385            weight=weight,
386            detail="vault is empty (no notes scanned)",
387            action=None,
388        )
389    affected = scan.notes_with_issues
390    raw = 100 * (1 - affected / total)
391    score = max(0, min(100, int(round(raw))))
392
393    if affected == 0:
394        detail = "all frontmatter valid"
395        action: str | None = None
396    else:
397        top_codes = sorted(scan.by_code.items(), key=lambda kv: -kv[1])[:3]
398        code_str = ", ".join(f"{c}={n}" for c, n in top_codes)
399        detail = f"{affected}/{total} note(s) with issues ({code_str})"
400        action = "uv run --no-project ~/.claude/skills/parsidion/scripts/vault_doctor.py --dry-run"
401    return DimensionScore(
402        name="metadata_quality",
403        score=score,
404        weight=weight,
405        detail=detail,
406        action=action,
407    )

410def score_embedding_coverage(vault: Path) -> DimensionScore:
411    """Embedding coverage: notes-with-embeddings ÷ total notes."""
412    weight = DIMENSION_WEIGHTS["embedding_coverage"]
413    db_path = get_embeddings_db_path(vault)
414    if not db_path.exists():
415        return DimensionScore(
416            name="embedding_coverage",
417            score=0,
418            weight=weight,
419            detail="embeddings.db absent — semantic search is unavailable",
420            action="uv run --no-project ~/.claude/skills/parsidion/scripts/build_embeddings.py",
421        )
422    try:
423        conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
424    except sqlite3.Error:
425        return DimensionScore(
426            name="embedding_coverage",
427            score=0,
428            weight=weight,
429            detail="embeddings.db unreadable",
430            action="uv run --no-project ~/.claude/skills/parsidion/scripts/build_embeddings.py",
431        )
432    try:
433        emb_row = conn.execute("SELECT COUNT(*) FROM note_embeddings").fetchone()
434        idx_row = conn.execute("SELECT COUNT(*) FROM note_index").fetchone()
435    except sqlite3.Error:
436        conn.close()
437        return DimensionScore(
438            name="embedding_coverage",
439            score=0,
440            weight=weight,
441            detail="embeddings.db schema is incomplete",
442            action="uv run --no-project ~/.claude/skills/parsidion/scripts/build_embeddings.py",
443        )
444    conn.close()
445
446    total = int(idx_row[0]) if idx_row else 0
447    embedded = int(emb_row[0]) if emb_row else 0
448    if total == 0:
449        return DimensionScore(
450            name="embedding_coverage",
451            score=0,
452            weight=weight,
453            detail="no notes indexed",
454            action=None,
455        )
456    missing = max(0, total - embedded)
457    score = int(round(100 * embedded / total))
458    if missing == 0:
459        detail = "all notes embedded"
460        action: str | None = None
461    else:
462        detail = f"{missing} of {total} note(s) unembedded"
463        action = (
464            "uv run --no-project ~/.claude/skills/parsidion/scripts/update_index.py"
465        )
466    return DimensionScore(
467        name="embedding_coverage",
468        score=score,
469        weight=weight,
470        detail=detail,
471        action=action,
472    )

475def _tag_pairs_near_duplicate(tags: list[str]) -> list[tuple[str, str]]:
476    """Detect near-duplicate tags: singular↔plural and Levenshtein-1 pairs.
477
478    Operates on lowercase forms. ``hook`` vs ``hooks`` is the canonical
479    false pair this catches; the audit also called out underscores but
480    those are a separate dimension (they fail the project's tag-convention
481    rule rather than duplicate an existing tag).
482    """
483    seen: set[str] = set()
484    pairs: list[tuple[str, str]] = []
485    sorted_tags = sorted(set(tags))
486    for a in sorted_tags:
487        if a in seen:
488            continue
489        for b in sorted_tags:
490            if a >= b:
491                continue
492            if _levenshtein1(a, b):
493                pairs.append((a, b))
494                seen.add(a)
495                seen.add(b)
496                break
497    return pairs
498

500def _levenshtein1(a: str, b: str) -> bool:
501    """True iff strings are edit-distance exactly 1 (single insert/delete/sub)."""
502    la, lb = len(a), len(b)
503    if abs(la - lb) > 1:
504        return False
505    if la == lb:
506        return sum(1 for x, y in zip(a, b, strict=True) if x != y) == 1
507    # Ensure a is the shorter one
508    if la > lb:
509        a, b = b, a
510        la, lb = lb, la
511    # Walk both; allow one insertion in b
512    i = j = edits = 0
513    while i < la and j < lb:
514        if a[i] == b[j]:
515            i += 1
516            j += 1
517        else:
518            edits += 1
519            if edits > 1:
520                return False
521            j += 1  # skip one char in the longer string
522    return True

525def score_tag_hygiene(vault: Path) -> DimensionScore:
526    """Tag hygiene: penalise singleton ratio, near-duplicates, underscores."""
527    weight = DIMENSION_WEIGHTS["tag_hygiene"]
528    conn = vault_metrics.open_db(vault)
529    if conn is None:
530        return DimensionScore(
531            name="tag_hygiene",
532            score=0,
533            weight=weight,
534            detail="note_index DB absent",
535            action="uv run --no-project ~/.claude/skills/parsidion/scripts/update_index.py",
536        )
537    try:
538        tags = vault_metrics.collect_tags(conn)
539    finally:
540        conn.close()
541    if not tags:
542        return DimensionScore(
543            name="tag_hygiene",
544            score=100,
545            weight=weight,
546            detail="no tags indexed",
547            action=None,
548        )
549
550    singletons = [t for t, n in tags if n == 1]
551    total_tags = len(tags)
552    singleton_ratio = len(singletons) / total_tags
553    near_dup_pairs = _tag_pairs_near_duplicate([t for t, _ in tags])
554    underscore_tags = [t for t, _ in tags if "_" in t]
555
556    # Singleton ratio penalty (capped at 40 — having niche tags isn't fatal),
557    # 8 per near-duplicate pair (capped at 40), 10 per underscore violation
558    # (capped at 30). Floor at 0.
559    raw = (
560        100
561        - min(40, int(round(singleton_ratio * 100)))
562        - min(40, len(near_dup_pairs) * 8)
563        - min(30, len(underscore_tags) * 10)
564    )
565    score = max(0, min(100, raw))
566
567    bits: list[str] = []
568    if singletons:
569        bits.append(f"{len(singletons)} singleton tag(s)")
570    if near_dup_pairs:
571        bits.append(f"{len(near_dup_pairs)} near-duplicate pair(s)")
572    if underscore_tags:
573        bits.append(f"{len(underscore_tags)} underscore tag(s)")
574    detail = "; ".join(bits) if bits else "tags are clean"
575    action = "vault-doctor --fix-tags" if (near_dup_pairs or underscore_tags) else None
576    return DimensionScore(
577        name="tag_hygiene",
578        score=score,
579        weight=weight,
580        detail=detail,
581        action=action,
582    )

585def score_file_hygiene(vault: Path) -> DimensionScore:
586    """File hygiene: unexpected modes + gitignored-but-tracked files."""
587    weight = DIMENSION_WEIGHTS["file_hygiene"]
588    # Mode check: vault .md files should be regular files (not symlinks to
589    # outside the vault, not world-writable). Group-writable is fine (vaults
590    # are often shared), but other-write or sticky is a misconfiguration.
591    bad_mode_files: list[str] = []
592    try:
593        for md in all_vault_notes_walk(vault):
594            try:
595                st = md.stat()
596            except OSError:
597                continue
598            mode = st.st_mode
599            if stat.S_ISLNK(mode):
600                # Symlinks inside the vault (e.g. Templates/) are legitimate;
601                # only flag if the link target escapes the vault root.
602                target = md.resolve(strict=False)
603                try:
604                    target.relative_to(vault.resolve())
605                except ValueError:
606                    bad_mode_files.append(str(md.relative_to(vault)))
607            elif mode & stat.S_IWOTH:
608                bad_mode_files.append(str(md.relative_to(vault)))
609    except Exception as exc:  # noqa: BLE001 — file walk must not crash the report
610        print(f"vault health file walk failed: {exc}", file=sys.stderr)
611        pass
612
613    # Tracked-but-gitignored check
614    tracked_gitignored: list[str] = []
615    try:
616        from doctor.scan import _git_tracked_gitignored
617
618        tracked_gitignored = _git_tracked_gitignored(vault)
619    except Exception as exc:  # noqa: BLE001
620        print(f"tracked-but-gitignored scan failed: {exc}", file=sys.stderr)
621        pass
622
623    # Deductions: 5 per finding class member, capped per class so a vault with
624    # 200 bad files still surfaces a number rather than pinning at 0 silently.
625    raw = 100 - min(50, len(bad_mode_files) * 5) - min(40, len(tracked_gitignored) * 5)
626    score = max(0, min(100, raw))
627
628    bits: list[str] = []
629    if bad_mode_files:
630        bits.append(f"{len(bad_mode_files)} file(s) with unexpected modes")
631    if tracked_gitignored:
632        bits.append(f"{len(tracked_gitignored)} tracked file(s) match .gitignore")
633    detail = "; ".join(bits) if bits else "file modes and git tracking are clean"
634    if tracked_gitignored:
635        action: str | None = "git -C <vault> rm --cached <files>"
636    elif bad_mode_files:
637        action = "chmod u+rwX,go-w <files>"
638    else:
639        action = None
640    return DimensionScore(
641        name="file_hygiene",
642        score=score,
643        weight=weight,
644        detail=detail,
645        action=action,

674def _build_warnings(note_types: dict[str, int]) -> list[str]:
675    """Surface structural issues the dimensions don't directly score.
676
677    Today: type-distribution underrepresentation. The audit found the
678    ``knowledge`` type was structurally absent because the summarizer
679    couldn't emit it; a type-distribution check would have flagged this
680    months earlier. Threshold: any of the nine valid types below 0.5% of
681    total notes (with a 100-note floor to avoid noisy small-vault warnings).
682    """
683    if not note_types:
684        return []
685    total = sum(note_types.values())
686    if total < 100:
687        return []
688    warnings: list[str] = []
689    # The nine types defined in doctor._state.VALID_TYPES — names hardwired
690    # here so this module stays stdlib-only and does not import doctor (which
691    # would pull vault_links and a heavier import graph).
692    valid_types = (
693        "pattern",
694        "debugging",
695        "research",
696        "project",
697        "daily",
698        "tool",
699        "language",
700        "framework",
701        "knowledge",
702    )
703    for t in valid_types:
704        n = note_types.get(t, 0)
705        if n == 0:
706            continue
707        ratio = n / total
708        if ratio < 0.005:  # under 0.5% of total
709            warnings.append(
710                f"'{t}' is underrepresented ({n} note(s), "
711                f"{ratio * 100:.2f}% of {total}) — the summarizer may not be routing to it"
712            )
713    return warnings

716def compute_health_report(
717    vault: Path | str | None = None, *, skip_metadata: bool = False
718) -> HealthReport:
719    """Compute the full vault health report.
720
721    Args:
722        vault: Optional vault path or name. Defaults to ``resolve_vault()``.
723        skip_metadata: When True, the metadata-quality scan is skipped (the
724            most expensive dimension on a large vault). The dimension is
725            reported with a neutral score and ``detail='skipped (--fast)'``
726            so the overall is not silently inflated. Used by ``vault-stats
727            --fast``.
728
729    Returns:
730        A populated ``HealthReport``. Never raises — per-dimension failures
731        degrade to low scores with detail strings (and metadata-scan failure
732        is wrapped so a broken ``check_note`` cannot abort the whole report).
733    """
734    resolved = resolve_vault(explicit=vault if isinstance(vault, str) else None)
735    # Pre-compute the metadata scan once so the dimension reuses it without
736    # walking the vault twice. Skipped under --fast so the report renders in
737    # well under a second on a 5k-note vault.
738    scan_summary = None
739    if not skip_metadata:
740        try:
741            from doctor.scan import scan_notes_readonly
742
743            scan_summary = scan_notes_readonly(resolved)
744        except Exception:  # noqa: BLE001 — degrade; dimension reports scan failed
745            scan_summary = None
746
747    if skip_metadata:
748        # Neutral score: include the dimension in the report (so the table
749        # still lists it) without letting it move the overall either way.
750        # The detail makes the skip visible so the user can re-run without
751        # --fast if they want the real number.
752        metadata_score = DimensionScore(
753            name="metadata_quality",
754            score=100,
755            weight=DIMENSION_WEIGHTS["metadata_quality"],
756            detail="skipped (--fast)",
757            action=None,
758        )
759    else:
760        metadata_score = score_metadata_quality(resolved, scan=scan_summary)
761
762    dimensions = [
763        score_index_freshness(resolved),
764        score_queue_health(resolved),
765        score_graph_connectivity(resolved),
766        metadata_score,
767        score_embedding_coverage(resolved),
768        score_tag_hygiene(resolved),
769        score_file_hygiene(resolved),
770    ]
771
772    total_weight = sum(d.weight for d in dimensions)
773    weighted = sum(d.score * d.weight for d in dimensions)
774    overall = int(round(weighted / total_weight)) if total_weight else 0
775
776    note_types = _note_type_distribution(resolved)
777    warnings = _build_warnings(note_types)
778
779    return HealthReport(
780        vault=resolved,
781        dimensions=dimensions,
782        overall=overall,
783        grade=_grade_for(overall),
784        note_types=note_types,
785        warnings=warnings,
786    )

311            score=0,
312            weight=weight,
313            detail="note_index DB absent — run update_index.py",

315        )

559    # (capped at 30). Floor at 0.
560    raw = (
561        100
562        - min(40, int(round(singleton_ratio * 100)))
563        - min(40, len(near_dup_pairs) * 8)
564        - min(30, len(underscore_tags) * 10)
565    )
566    score = max(0, min(100, raw))
567
568    bits: list[str] = []
569    if singletons:
570        bits.append(f"{len(singletons)} singleton tag(s)")
571    if near_dup_pairs:
572        bits.append(f"{len(near_dup_pairs)} near-duplicate pair(s)")
573    if underscore_tags:
574        bits.append(f"{len(underscore_tags)} underscore tag(s)")
575    detail = "; ".join(bits) if bits else "tags are clean"
576    action = "vault-doctor --fix-tags" if (near_dup_pairs or underscore_tags) else None
577    return DimensionScore(
578        name="tag_hygiene",
579        score=score,
580        weight=weight,
581        detail=detail,
582        action=action,
583    )

def score_file_hygiene(vault: pathlib.Path) -> DimensionScore: 
586def score_file_hygiene(vault: Path) -> DimensionScore:
587    """File hygiene: unexpected modes + gitignored-but-tracked files."""
588    weight = DIMENSION_WEIGHTS["file_hygiene"]
589    # Mode check: vault .md files should be regular files (not symlinks to
590    # outside the vault, not world-writable). Group-writable is fine (vaults
591    # are often shared), but other-write or sticky is a misconfiguration.
592    bad_mode_files: list[str] = []
593    try:
594        for md in all_vault_notes_walk(vault):
595            try:
596                st = md.stat()
597            except OSError:
598                continue
599            mode = st.st_mode
600            if stat.S_ISLNK(mode):
601                # Symlinks inside the vault (e.g. Templates/) are legitimate;
602                # only flag if the link target escapes the vault root.
603                target = md.resolve(strict=False)
604                try:
605                    target.relative_to(vault.resolve())
606                except ValueError:
607                    bad_mode_files.append(str(md.relative_to(vault)))
608            elif mode & stat.S_IWOTH:
609                bad_mode_files.append(str(md.relative_to(vault)))
610    except Exception as exc:  # noqa: BLE001 — file walk must not crash the report
611        print(f"vault health file walk failed: {exc}", file=sys.stderr)
612        pass
613
614    # Tracked-but-gitignored check
615    tracked_gitignored: list[str] = []
616    try:
617        from doctor.scan import _git_tracked_gitignored
618
619        tracked_gitignored = _git_tracked_gitignored(vault)
620    except Exception as exc:  # noqa: BLE001
621        print(f"tracked-but-gitignored scan failed: {exc}", file=sys.stderr)
622        pass
623
624    # Deductions: 5 per finding class member, capped per class so a vault with
625    # 200 bad files still surfaces a number rather than pinning at 0 silently.
626    raw = 100 - min(50, len(bad_mode_files) * 5) - min(40, len(tracked_gitignored) * 5)
627    score = max(0, min(100, raw))
628
629    bits: list[str] = []
630    if bad_mode_files:
631        bits.append(f"{len(bad_mode_files)} file(s) with unexpected modes")
632    if tracked_gitignored:
633        bits.append(f"{len(tracked_gitignored)} tracked file(s) match .gitignore")
634    detail = "; ".join(bits) if bits else "file modes and git tracking are clean"
635    if tracked_gitignored:
636        action: str | None = "git -C <vault> rm --cached <files>"
637    elif bad_mode_files:
638        action = "chmod u+rwX,go-w <files>"
639    else:
640        action = None
641    return DimensionScore(
642        name="file_hygiene",
643        score=score,
644        weight=weight,
645        detail=detail,
646        action=action,
647    )

717def compute_health_report(
718    vault: Path | str | None = None, *, skip_metadata: bool = False
719) -> HealthReport:
720    """Compute the full vault health report.
721
722    Args:
723        vault: Optional vault path or name. Defaults to ``resolve_vault()``.
724        skip_metadata: When True, the metadata-quality scan is skipped (the
725            most expensive dimension on a large vault). The dimension is
726            reported with a neutral score and ``detail='skipped (--fast)'``
727            so the overall is not silently inflated. Used by ``vault-stats
728            --fast``.
729
730    Returns:
731        A populated ``HealthReport``. Never raises — per-dimension failures
732        degrade to low scores with detail strings (and metadata-scan failure
733        is wrapped so a broken ``check_note`` cannot abort the whole report).
734    """
735    resolved = resolve_vault(explicit=vault if isinstance(vault, str) else None)
736    # Pre-compute the metadata scan once so the dimension reuses it without
737    # walking the vault twice. Skipped under --fast so the report renders in
738    # well under a second on a 5k-note vault.
739    scan_summary = None
740    if not skip_metadata:
741        try:
742            from doctor.scan import scan_notes_readonly
743
744            scan_summary = scan_notes_readonly(resolved)
745        except Exception:  # noqa: BLE001 — degrade; dimension reports scan failed
746            scan_summary = None
747
748    if skip_metadata:
749        # Neutral score: include the dimension in the report (so the table
750        # still lists it) without letting it move the overall either way.
751        # The detail makes the skip visible so the user can re-run without
752        # --fast if they want the real number.
753        metadata_score = DimensionScore(
754            name="metadata_quality",
755            score=100,
756            weight=DIMENSION_WEIGHTS["metadata_quality"],
757            detail="skipped (--fast)",
758            action=None,
759        )
760    else:
761        metadata_score = score_metadata_quality(resolved, scan=scan_summary)
762
763    dimensions = [
764        score_index_freshness(resolved),
765        score_queue_health(resolved),
766        score_graph_connectivity(resolved),
767        metadata_score,
768        score_embedding_coverage(resolved),
769        score_tag_hygiene(resolved),
770        score_file_hygiene(resolved),
771    ]
772
773    total_weight = sum(d.weight for d in dimensions)
774    weighted = sum(d.score * d.weight for d in dimensions)
775    overall = int(round(weighted / total_weight)) if total_weight else 0
776
777    note_types = _note_type_distribution(resolved)
778    warnings = _build_warnings(note_types)
779
780    return HealthReport(
781        vault=resolved,
782        dimensions=dimensions,
783        overall=overall,
784        grade=_grade_for(overall),
785        note_types=note_types,
786        warnings=warnings,
787    )
