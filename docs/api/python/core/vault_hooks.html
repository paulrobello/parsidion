core.vault_hooks

Hook utilities, event logging, environment management, and transcript analysis.

Provides hook event logging, safe environment construction for child processes, transcript text extraction, project name detection, process checking, and transcript category detection/parsing shared by session_stop and subagent_stop hooks.

This module is part of the vault_common split (ARC-005). All public symbols are re-exported from vault_common for backward compatibility.

 68def log_hook_error(hook_name: str) -> None:
 69    """Append a timestamped traceback entry to the persistent hook error log.
 70
 71    QA-003: the single implementation of what was copy-pasted as
 72    ``_log_hook_error`` into five hook scripts. Called only from a hook's
 73    outermost ``except Exception`` handler so that unexpected programming
 74    errors (regressions, NameErrors, etc.) are written to
 75    ``~/.claude/logs/parsidion-hook-errors.log`` rather than disappearing
 76    into stderr. Best-effort — never raises.
 77
 78    Args:
 79        hook_name: Short identifier for the hook (e.g. ``"session_stop_hook"``).
 80    """
 81    error_log = secure_log_dir() / "parsidion-hook-errors.log"
 82    try:
 83        ts = datetime.now().isoformat(timespec="seconds")
 84        tb = traceback.format_exc()
 85        entry = f"[{ts}] {hook_name}\n{tb}\n"
 86        rotate_log_file(error_log)
 87        with open(error_log, "a", encoding="utf-8") as fh:
 88            fh.write(entry)
 89    except Exception as exc:  # noqa: BLE001 — logging must never raise
 90        print(f"hook error log write failed: {exc}", file=sys.stderr)

184def _coerce_env_value(value: object) -> str | None:
185    """Convert a config value into a process env string.
186
187    Empty strings and explicit ``null`` values are treated as unset.
188    """
189    if value is None:
190        return None
191    if isinstance(value, str):
192        return value if value.strip() else None
193    return str(value)

196def _configured_env_defaults(vault: Path | None = None) -> dict[str, str]:
197    """Return supported env defaults from ``vault/config.yaml``.
198
199    Environment variables are represented in config under the ``anthropic_env``
200    section using their real env var names as keys.
201
202    SEC-007: network-affecting keys (see ``_NETWORK_AFFECTING_ENV_KEYS``) are
203    honored only when their effective value comes from ``config.local.yaml``
204    or when ``config.yaml`` is not git-tracked in the vault repo — a synced
205    ``config.yaml`` must not be able to redirect where requests and auth
206    headers are sent. Refused keys are reported once on stderr.
207    """
208    global _untrusted_network_env_warned
209
210    config = load_config(vault=vault)
211    section = config.get("anthropic_env")
212    if not isinstance(section, dict):
213        return {}
214
215    has_network_keys = any(key in section for key in _NETWORK_AFFECTING_ENV_KEYS)
216    sources: dict[tuple[str, str], str] = (
217        config_key_sources(vault=vault) if has_network_keys else {}
218    )
219    # Lazily computed: True only when the vault is a git repo AND config.yaml
220    # is not gitignored there (i.e. the file syncs to a remote).
221    tracked_config_yaml: bool | None = None
222
223    resolved: dict[str, str] = {}
224    refused: list[str] = []
225    for key in _CONFIGURABLE_ENV_KEYS:
226        if key not in section:
227            continue
228        value = _coerce_env_value(section[key])
229        if value is None:
230            continue
231        if (
232            key in _NETWORK_AFFECTING_ENV_KEYS
233            and sources.get(("anthropic_env", key)) != "config.local.yaml"
234        ):
235            if tracked_config_yaml is None:
236                vault_dir = vault if vault is not None else resolve_vault()
237                tracked_config_yaml = (vault_dir / ".git").exists() and (
238                    not _git_path_ignored("config.yaml", vault_dir)
239                )
240            if tracked_config_yaml:
241                refused.append(key)
242                continue
243        resolved[key] = value
244
245    if refused and not _untrusted_network_env_warned:
246        _untrusted_network_env_warned = True
247        print(
248            "vault_hooks: ignoring anthropic_env network keys "
249            f"{', '.join(sorted(refused))} from tracked config.yaml "
250            "(honored only from config.local.yaml or a gitignored "
251            "config.yaml); SEC-007",
252            file=sys.stderr,
253        )
254    return resolved

257def apply_configured_env_defaults(vault: Path | None = None) -> None:
258    """Populate missing process env vars from ``vault/config.yaml``.
259
260    Existing environment variables always win over config values.
261    Call this before SDK-based Claude usage that reads from ``os.environ``
262    directly instead of an explicit ``env=`` subprocess mapping.
263    """
264    for key, value in _configured_env_defaults(vault=vault).items():
265        os.environ.setdefault(key, value)

268def env_without_claudecode(vault: Path | None = None) -> dict[str, str]:
269    """Return a filtered copy of the current environment for child processes.
270
271    Only includes variables listed in ``_SAFE_ENV_KEYS``, which avoids leaking
272    secrets or triggering the Claude nesting guard (``CLAUDECODE``).
273    Missing supported Anthropic-compatible variables are filled from the vault
274    config's ``anthropic_env`` section when present.
275
276    Always injects ``PARSIDION_INTERNAL=1`` so that hook scripts invoked by the
277    resulting ``claude -p`` session can detect and skip internal sessions.
278
279    Returns:
280        A dict suitable for passing as ``env=`` to ``subprocess.run`` / ``Popen``.
281    """
282    env = {k: v for k, v in os.environ.items() if k in _SAFE_ENV_KEYS}
283    for key, value in _configured_env_defaults(vault=vault).items():
284        env.setdefault(key, value)
285    env["PARSIDION_INTERNAL"] = "1"
286    return env

294def extract_text_from_content(content: object) -> str:
295    """Extract plain text from a transcript message content field.
296
297    Content can be a plain string or an array of content blocks (each with
298    a ``type`` and ``text`` field for text blocks).
299
300    Args:
301        content: The message content -- typically a string or list of blocks.
302
303    Returns:
304        Concatenated text from all text blocks.
305    """
306    if isinstance(content, str):
307        return content
308    if isinstance(content, list):
309        parts: list[str] = []
310        for block in content:
311            if isinstance(block, dict) and block.get("type") == "text":
312                text = block.get("text", "")
313                if isinstance(text, str):
314                    parts.append(text)
315        return "\n".join(parts)
316    return ""

329def allowed_transcript_roots(cwd: str | None = None) -> list[Path]:
330    """Return allowed root directories for transcript files.
331
332    Supports Claude Code, pi, Codex, and Gemini transcript locations:
333
334    - ``~/.claude/`` (Claude Code transcripts)
335    - ``~/.pi/`` (pi global transcripts, e.g. ``~/.pi/agent/sessions``)
336    - ``<cwd>/.pi/`` (project-local pi transcripts, e.g. ``.pi/agent-sessions``)
337    - ``$CODEX_HOME/sessions`` or ``~/.codex/sessions`` (Codex transcripts)
338    - ``$GEMINI_HOME`` or ``~/.gemini`` (user Gemini transcripts)
339    - ``<cwd>/.gemini/`` (project-local Gemini transcripts)
340
341    Args:
342        cwd: Optional working directory for project-local ``.pi``/``.gemini`` roots.
343
344    Returns:
345        De-duplicated list of resolved root paths.
346    """
347    roots: list[Path] = [
348        Path.home() / ".claude",
349        Path.home() / ".pi",
350        codex_home() / "sessions",
351        gemini_home(),
352    ]
353
354    if cwd:
355        try:
356            cwd_path = Path(cwd).resolve()
357            roots.append(cwd_path / ".pi")
358            roots.append(cwd_path / ".gemini")
359        except OSError:
360            pass
361
362    deduped: list[Path] = []
363    seen: set[str] = set()
364    for root in roots:
365        try:
366            resolved = root.resolve()
367        except OSError:
368            resolved = root
369        key = str(resolved)
370        if key not in seen:
371            seen.add(key)
372            deduped.append(resolved)
373
374    return deduped

377def is_allowed_transcript_path(transcript_path: Path, cwd: str | None = None) -> bool:
378    """Return True when *transcript_path* is inside an allowed transcript root.
379
380    SEC-010: Also requires the ``.jsonl`` suffix so that non-transcript files
381    (e.g. ``settings.json``, ``CLAUDE.md``) cannot be read even if they reside
382    under an allowed root such as ``~/.claude/``.
383    """
384    # SEC-010: Reject anything that is not a .jsonl file to prevent hook JSON
385    # from pointing at non-transcript files (settings, keys, etc.) under ~/.claude.
386    if transcript_path.suffix != ".jsonl":
387        return False
388
389    try:
390        resolved = transcript_path.resolve()
391    except OSError:
392        return False
393
394    for root in allowed_transcript_roots(cwd=cwd):
395        try:
396            if resolved.is_relative_to(root):
397                return True
398        except ValueError:
399            continue
400
401    return False

468def get_project_name(cwd: str | None = None) -> str:
469    """Extract a project name from *cwd* or the current directory.
470
471    Uses the basename of the directory. If the directory is inside a git
472    repository, uses the repository root basename instead.
473    """
474    if cwd is None:
475        cwd = os.getcwd()
476
477    path = Path(cwd).resolve()
478
479    # Walk up to find a .git directory
480    check = path
481    while check != check.parent:
482        if (check / ".git").exists():
483            return check.name
484        check = check.parent
485
486    # Fallback: basename of the given directory
487    return path.name

495def is_process_running(pid: int) -> bool:
496    """Return True if a process with *pid* is currently running.
497
498    Uses ``os.kill(pid, 0)`` which sends no signal but checks process existence.
499
500    QA-007: Canonical implementation shared by update_index.py and vault_doctor.py.
501
502    SEC-016: PermissionError now returns False. A PID we cannot signal
503    belongs to another user, which under parsidion's single-user threat
504    model means it is not one of our stale processes — the old True made
505    a leftover ``pid: 1`` permanently block every PID-file singleton guard
506    (doctor runs were stuck until the state file was hand-edited). Callers
507    that need real mutual exclusion use ``vault_fs.try_singleton_lock``
508    (flock), not this probe.
509
510    Args:
511        pid: Process ID to check.
512
513    Returns:
514        True if the process is running and signalable by us.
515    """
516    try:
517        os.kill(pid, 0)
518        return True
519    except ProcessLookupError:
520        return False
521    except PermissionError:
522        # Cannot signal -> not ours -> report not running (SEC-016).
523        return False

576def parse_transcript_lines(lines: list[str]) -> list[str]:
577    """Parse JSONL transcript lines and extract assistant message text.
578
579    Supports both Claude Code transcript events (``type: assistant``) and
580    pi transcript events (``type: message`` with ``message.role=assistant``).
581
582    Args:
583        lines: Raw JSONL lines from the transcript file.
584
585    Returns:
586        A list of text strings from assistant messages.
587    """
588    assistant_texts: list[str] = []
589    for raw_line in lines:
590        line = raw_line.strip()
591        if not line:
592            continue
593        try:
594            entry = json.loads(line)
595        except (json.JSONDecodeError, ValueError):
596            continue
597
598        role: str | None = None
599        content: object = None
600
601        message = entry.get("message")
602        if isinstance(message, dict):
603            role_raw = message.get("role")
604            if isinstance(role_raw, str):
605                role = role_raw
606            content = message.get("content")
607
608        if role is None:
609            msg_type = entry.get("type")
610            if isinstance(msg_type, str) and msg_type in {"assistant", "user"}:
611                role = msg_type
612                content = entry.get("content")
613
614        if role != "assistant" or content is None:
615            continue
616
617        text = extract_text_from_content(content)
618        if text.strip():
619            assistant_texts.append(text)
620
621    return assistant_texts

624def parse_codex_transcript_lines(lines: list[str]) -> list[str]:
625    """Parse Codex rollout JSONL lines and extract assistant message text."""
626    texts: list[str] = []
627    for line in lines:
628        try:
629            record = json.loads(line)
630        except (json.JSONDecodeError, TypeError):
631            continue
632
633        item = None
634        if isinstance(record, dict):
635            item = record.get("payload")
636            if not isinstance(item, dict):
637                item = record.get("item")
638        if not isinstance(item, dict):
639            continue
640        if item.get("type") != "message" or item.get("role") != "assistant":
641            continue
642
643        content = item.get("content", [])
644        if isinstance(content, str):
645            if content.strip():
646                texts.append(content.strip())
647            continue
648        if not isinstance(content, list):
649            continue
650
651        chunks: list[str] = []
652        for block in content:
653            if not isinstance(block, dict):
654                continue
655            if block.get("type") in {"output_text", "text"}:
656                text = block.get("text")
657                if isinstance(text, str) and text.strip():
658                    chunks.append(text.strip())
659        if chunks:
660            texts.append("\n".join(chunks))
661
662    return texts

411    hook: str,
412    project: str,

418

420    (default: ``true``). Rotates (keeps last *max_lines*) when the file
421    exceeds ``event_log.max_lines`` (default: 10 000).
422
423    Args:
424        hook: Hook name, e.g. ``"SessionEnd"``.
425        project: Project name.
426        duration_ms: Hook wall-clock time in milliseconds.
427        vault: Optional vault path. Defaults to resolve_vault().
428        **extra: Additional key-value pairs to include in the event object.
429    """

431    if not get_config("event_log", "enabled", True):
432        return
433

437        "project": project,
438        "duration_ms": round(duration_ms, 1),
439    }
440    event.update(extra)
441

443    # ARC-011: honour ``event_log.path`` from config (absolute path override).
444    # When set to a non-empty string, use it instead of the vault-relative
445    # default. None / empty falls through to ``<vault>/hook_events.log``.
446    # SEC-006: the override must resolve inside the vault or inside
447    # ``~/.claude/logs`` — anything else (e.g. ``~/.bashrc``) is refused with
448    # a one-time warning and the vault-relative default is used, so a synced

456            candidate, secure_log_dir()

463                "[parsidion] event_log.path is outside the vault and ~/.claude/logs; "
464                f"ignoring '{configured_path}'",
465                file=sys.stderr,
466            )

469    )
470
471    try:
472        vault.mkdir(parents=True, exist_ok=True)
473        log_path.parent.mkdir(parents=True, exist_ok=True)
474        line = json.dumps(event) + "\n"
475
476        # SEC-008: Create the log file with mode 0o600 (owner read/write only)
477        # so hook events are not world-readable. os.open with O_CREAT | O_RDWR
478        # sets the mode atomically on first creation; subsequent opens inherit
479        # the existing file mode unchanged (chmod is not called on existing files

487        )

496                    # Rotate via tmp + atomic replace (still holding the lock
497                    # on the original handle) so a crash mid-rotation cannot
498                    # truncate the log to a partial state.

505                        out.writelines(keep)
506                        out.write(line)
507                    tmp.replace(log_path)
508                else:
509                    f.seek(0, 2)
510                    f.write(line)
511            finally:
512                funlock(f)
513    except OSError:
514        pass
