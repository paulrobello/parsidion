core.vault_metrics

vault_metrics — stdlib-only data layer for vault analytics.

Provides pure data-gathering functions over the note_index database and the vault file system. No third-party dependencies (rich, fastembed, sqlite-vec).

This module is used by vault_stats (the CLI display layer) and may be imported directly by other vault tools or tests without requiring the [tools] optional-dependency group.

ARC-007: separating the data/query layer from the display/CLI layer so vault_stats can be imported as a library without triggering an ImportError when rich is not installed.

 57def open_db(vault: Path | None = None) -> sqlite3.Connection | None:
 58    """Open the embeddings.db in read-only mode.
 59
 60    Args:
 61        vault: Optional vault path. Defaults to resolve_vault().
 62
 63    Returns:
 64        An open connection, or None if the DB is absent, unreadable, or
 65        lacks the note_index table. A present-but-empty file (the 0-byte
 66        embeddings.db the parsight search backend keeps on disk) opens as a
 67        valid empty database, so the table is probed explicitly to route
 68        that case to the same no-DB fallback as a missing file.
 69    """
 70    db_path = get_embeddings_db_path(vault)
 71    if not db_path.exists():
 72        return None
 73    try:
 74        conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
 75        conn.row_factory = sqlite3.Row
 76    except sqlite3.Error:
 77        return None
 78    try:
 79        has_note_index = conn.execute(
 80            "SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'note_index'"
 81        ).fetchone()
 82    except sqlite3.Error:
 83        conn.close()
 84        return None
 85    if has_note_index is None:
 86        conn.close()
 87        return None
 88    return conn

 99def connect_with_vec(db_path: Path) -> sqlite3.Connection:
100    """Open *db_path* read-write with the sqlite-vec extension loaded.
101
102    QA-017: the single sqlite-vec-loading connector, previously
103    copy-pasted in build_embeddings.open_db and
104    cli.search.embeddings._open_db_semantic. The optional import lives in
105    this branch so the stdlib-only gate stays green for callers that
106    never touch the vector path.
107
108    Args:
109        db_path: Path to the SQLite database file.
110
111    Returns:
112        An open sqlite3.Connection with sqlite-vec loaded.
113
114    Raises:
115        VecExtensionMissing: sqlite-vec is not installed.
116    """
117    try:
118        import sqlite_vec  # type: ignore[import-untyped]
119    except ImportError as exc:
120        raise VecExtensionMissing(
121            "sqlite-vec not installed — run: uv tool install --editable '.[tools]'"
122        ) from exc
123    conn = sqlite3.connect(db_path)
124    conn.enable_load_extension(True)
125    sqlite_vec.load(conn)
126    conn.enable_load_extension(False)
127    return conn

130def fetch_all(
131    conn: sqlite3.Connection, sql: str, params: tuple = ()
132) -> list[sqlite3.Row]:
133    """Execute *sql* and return all rows.
134
135    Args:
136        conn: Open DB connection.
137        sql: SQL query string.
138        params: Query parameters.
139
140    Returns:
141        List of Row objects.
142    """
143    try:
144        return conn.execute(sql, params).fetchall()
145    except sqlite3.Error:
146        return []

154def collect_summary(conn: sqlite3.Connection) -> dict:
155    """Return note counts by folder and by type.
156
157    Args:
158        conn: Open DB connection.
159
160    Returns:
161        Dict with keys: total (int), by_folder (list of {folder, n}),
162        by_type (list of {note_type, n}).
163    """
164    total = fetch_all(conn, "SELECT COUNT(*) AS n FROM note_index")[0]["n"]
165    folder_rows = fetch_all(
166        conn,
167        "SELECT folder, COUNT(*) AS n FROM note_index GROUP BY folder ORDER BY n DESC",
168    )
169    type_rows = fetch_all(
170        conn,
171        "SELECT note_type, COUNT(*) AS n FROM note_index GROUP BY note_type ORDER BY n DESC",
172    )
173    return {
174        "total": total,
175        "by_folder": [dict(r) for r in folder_rows],
176        "by_type": [dict(r) for r in type_rows],
177    }

180def collect_stale(conn: sqlite3.Connection) -> list[dict]:
181    """Return notes flagged as stale.
182
183    Args:
184        conn: Open DB connection.
185
186    Returns:
187        List of dicts with keys: stem, title, folder, mtime, age (YYYY-MM-DD str).
188    """
189    rows = fetch_all(
190        conn,
191        "SELECT stem, title, folder, mtime FROM note_index WHERE is_stale = 1 ORDER BY mtime ASC",
192    )
193    results = []
194    for row in rows:
195        try:
196            dt = datetime.fromtimestamp(row["mtime"], tz=UTC)
197            age = dt.strftime("%Y-%m-%d")
198        except (OSError, ValueError):
199            age = "unknown"
200        results.append(
201            {
202                "stem": row["stem"],
203                "title": row["title"],
204                "folder": row["folder"],
205                "mtime": row["mtime"],
206                "age": age,
207            }
208        )
209    return results

258def collect_growth(conn: sqlite3.Connection, weeks: int = 8) -> list[dict]:
259    """Return notes created per week for the last N weeks.
260
261    Args:
262        conn: Open DB connection.
263        weeks: Number of weeks to cover.
264
265    Returns:
266        List of dicts with keys: label (str), n (int), week_num (int),
267        ordered from oldest to most-recent.
268    """
269    now = time.time()
270    week_secs = 7 * 24 * 3600
271    cutoff = now - weeks * week_secs
272
273    rows = fetch_all(
274        conn,
275        "SELECT mtime FROM note_index WHERE mtime >= ? ORDER BY mtime ASC",
276        (cutoff,),
277    )
278
279    buckets: dict[int, int] = {}
280    for row in rows:
281        week_num = int((now - row["mtime"]) / week_secs)
282        week_num = min(week_num, weeks - 1)
283        buckets[week_num] = buckets.get(week_num, 0) + 1
284
285    result = []
286    for w in range(weeks - 1, -1, -1):
287        n = buckets.get(w, 0)
288        label = "this week" if w == 0 else f"{w}w ago"
289        result.append({"label": label, "n": n, "week_num": w})
290    return result

293def collect_tags(conn: sqlite3.Connection) -> list[tuple[str, int]]:
294    """Collect all tags from note_index, returning (tag, count) sorted by count desc.
295
296    The ``tags`` column stores either a comma-separated string
297    (``"python, vault, hooks"``) or a JSON array (``["python", "vault"]``).
298    Both formats are handled; malformed values are skipped silently.
299
300    Args:
301        conn: Open DB connection.
302
303    Returns:
304        List of (tag, count) tuples sorted by count descending.
305    """
306    rows = fetch_all(
307        conn, "SELECT tags FROM note_index WHERE tags IS NOT NULL AND tags != ''"
308    )
309    counts: dict[str, int] = {}
310    for row in rows:
311        raw = row["tags"]
312        try:
313            parsed = json.loads(raw)
314            tag_list: list[str] = parsed if isinstance(parsed, list) else []
315        except (json.JSONDecodeError, TypeError):
316            tag_list = [t.strip() for t in raw.split(",")]
317        for tag in tag_list:
318            t = str(tag).strip()
319            if t:
320                counts[t] = counts.get(t, 0) + 1
321    return sorted(counts.items(), key=lambda x: -x[1])

324def collect_graph(conn: sqlite3.Connection) -> dict:
325    """Return knowledge graph analytics from the note_index.
326
327    Args:
328        conn: Open DB connection.
329
330    Returns:
331        Dict with keys: total, total_links, avg_links, linked_count,
332        unlinked_count, hub_notes (list of dicts), isolated_notes (list of dicts).
333        Retrieval-readiness keys (graph-expansion feature): expandable_count
334        (notes carrying >=1 related link), avg_related_per_note, total_targets
335        (sum of related stems across all notes), dangling_targets (related stems
336        that resolve to no indexed note and are therefore skipped at retrieval).
337    """
338    all_rows = fetch_all(
339        conn,
340        "SELECT stem, title, folder, incoming_links, related FROM note_index",
341    )
342    if not all_rows:
343        return {
344            "total": 0,
345            "total_links": 0,
346            "avg_links": 0.0,
347            "linked_count": 0,
348            "unlinked_count": 0,
349            "hub_notes": [],
350            "isolated_notes": [],
351            "expandable_count": 0,
352            "avg_related_per_note": 0.0,
353            "total_targets": 0,
354            "dangling_targets": 0,
355        }
356
357    total = len(all_rows)
358    total_links = sum(r["incoming_links"] or 0 for r in all_rows)
359    avg_links = total_links / total if total else 0.0
360    linked_count = sum(1 for r in all_rows if (r["incoming_links"] or 0) > 0)
361    unlinked_count = total - linked_count
362
363    hub_rows = sorted(
364        [r for r in all_rows if (r["incoming_links"] or 0) >= 5],
365        key=lambda r: -(r["incoming_links"] or 0),
366    )[:10]
367
368    isolated_rows = [
369        r
370        for r in all_rows
371        if (r["incoming_links"] or 0) == 0 and not (r["related"] or "").strip()
372    ]
373
374    # Retrieval-readiness metrics for the graph-expansion feature.
375    valid_stems = {r["stem"] for r in all_rows}
376    expandable_count = sum(1 for r in all_rows if (r["related"] or "").strip())
377    total_targets = 0
378    dangling_targets = 0
379    for r in all_rows:
380        for target in parse_related_stems(r["related"] or ""):
381            total_targets += 1
382            if target not in valid_stems:
383                dangling_targets += 1
384    avg_related_per_note = total_targets / total if total else 0.0
385
386    return {
387        "total": total,
388        "total_links": total_links,
389        "avg_links": avg_links,
390        "linked_count": linked_count,
391        "unlinked_count": unlinked_count,
392        "hub_notes": [dict(r) for r in hub_rows],
393        "isolated_notes": [dict(r) for r in isolated_rows],
394        "expandable_count": expandable_count,
395        "avg_related_per_note": avg_related_per_note,
396        "total_targets": total_targets,
397        "dangling_targets": dangling_targets,
398    }

401def collect_pending(vault: Path | None = None) -> dict:
402    """Return a summary of pending_summaries.jsonl queue.
403
404    Args:
405        vault: Optional vault path. Defaults to resolve_vault().
406
407    Returns:
408        Dict with keys: exists (bool), total (int), source_counts (dict),
409        project_counts (dict), oldest_ts (str|None), token_estimate (int),
410        entries (list of dicts).
411    """
412    vault = vault or resolve_vault()
413    pending_path = vault / "pending_summaries.jsonl"
414    if not pending_path.exists():
415        return {
416            "exists": False,
417            "total": 0,
418            "source_counts": {},
419            "project_counts": {},
420            "oldest_ts": None,
421            "token_estimate": 0,
422            "entries": [],
423        }
424
425    entries: list[dict] = []
426    try:
427        with open(pending_path, encoding="utf-8") as fh:
428            for line in fh:
429                line = line.strip()
430                if line:
431                    try:
432                        entries.append(json.loads(line))
433                    except json.JSONDecodeError:
434                        pass
435    except OSError:
436        return {
437            "exists": True,
438            "error": True,
439            "total": 0,
440            "source_counts": {},
441            "project_counts": {},
442            "oldest_ts": None,
443            "token_estimate": 0,
444            "entries": [],
445        }
446
447    total = len(entries)
448    source_counts: dict[str, int] = {}
449    project_counts: dict[str, int] = {}
450    oldest_ts: str | None = None
451
452    for entry in entries:
453        src = entry.get("source", "session")
454        source_counts[src] = source_counts.get(src, 0) + 1
455        project = entry.get("project", "")
456        if project:
457            project_counts[project] = project_counts.get(project, 0) + 1
458        ts = entry.get("timestamp", "")
459        if ts and (oldest_ts is None or ts < oldest_ts):
460            oldest_ts = ts
461
462    return {
463        "exists": True,
464        "total": total,
465        "source_counts": source_counts,
466        "project_counts": project_counts,
467        "oldest_ts": oldest_ts,
468        "token_estimate": total * 100,
469        "entries": entries,
470    }

473def collect_dead_letters(vault: Path | None = None) -> dict:
474    """Return a summary of dead_letters.jsonl (purged dead-letter entries).
475
476    Entries are written by summarize_sessions.remove_processed() when a
477    pending queue entry hits _MAX_ATTEMPTS and is purged.
478
479    Args:
480        vault: Optional vault path. Defaults to resolve_vault().
481
482    Returns:
483        Dict with keys: exists (bool), total (int), recent (list of up to
484        3 dicts with project/last_failure/dead_lettered_at, most recent
485        first).
486    """
487    vault = vault or resolve_vault()
488    dead_letter_path = vault / "dead_letters.jsonl"
489    if not dead_letter_path.exists():
490        return {"exists": False, "total": 0, "recent": []}
491
492    entries: list[dict] = []
493    try:
494        with open(dead_letter_path, encoding="utf-8") as fh:
495            for line in fh:
496                line = line.strip()
497                if line:
498                    try:
499                        entries.append(json.loads(line))
500                    except json.JSONDecodeError:
501                        pass
502    except OSError:
503        return {"exists": True, "error": True, "total": 0, "recent": []}
504
505    recent = [
506        {
507            "project": e.get("project", "unknown"),
508            "last_failure": e.get("last_failure", ""),
509            "dead_lettered_at": e.get("dead_lettered_at", ""),
510        }
511        for e in entries[-3:]
512    ]
513    recent.reverse()
514
515    return {"exists": True, "total": len(entries), "recent": recent}

553def collect_timeline(
554    conn: sqlite3.Connection | None, days: int = 30, vault: Path | None = None
555) -> list[dict]:
556    """Return per-day note counts for the last N days.
557
558    Args:
559        conn: Open DB connection, or None for file-walk fallback.
560        days: Number of days to cover.
561        vault: Optional vault path. Defaults to resolve_vault().
562
563    Returns:
564        List of dicts with keys: date (YYYY-MM-DD str), n (int), is_today (bool),
565        ordered from oldest to most-recent (index 0 = days-1 ago).
566    """
567    from datetime import date, timedelta
568
569    # Bucket by local calendar day (not rolling 24h windows): bucket index d
570    # holds notes whose mtime falls on calendar day (today - d), matching the
571    # calendar-date labels below regardless of the time of day the report runs.
572    today = date.today()
573    oldest_day = today - timedelta(days=days - 1)
574    # Local midnight at the start of the oldest bucketed day.
575    cutoff_ts = datetime(oldest_day.year, oldest_day.month, oldest_day.day).timestamp()
576
577    day_counts: dict[int, int] = {i: 0 for i in range(days)}
578
579    def _age_days(mtime: float) -> int:
580        age = (today - date.fromtimestamp(mtime)).days
581        return min(max(age, 0), days - 1)
582
583    if conn is not None:
584        rows = fetch_all(
585            conn,
586            "SELECT mtime FROM note_index WHERE mtime >= ?",
587            (cutoff_ts,),
588        )
589        for row in rows:
590            age_days = _age_days(row["mtime"])
591            day_counts[age_days] = day_counts.get(age_days, 0) + 1
592    else:
593        # ENH-004: route through the shared all_vault_notes() helper so this
594        # no-DB walk gets the same EXCLUDE_DIRS / CLAUDE.md / TAGS.md /
595        # MANIFEST.md / SEC-106 symlink filtering as every other read path,
596        # instead of a raw rglob that silently includes them.
597        vault = vault or resolve_vault()
598        if vault.exists():
599            for md in all_vault_notes(vault):
600                try:
601                    mtime = md.stat().st_mtime
602                except OSError:
603                    continue
604                if mtime < cutoff_ts:
605                    continue
606                age_days = _age_days(mtime)
607                day_counts[age_days] = day_counts.get(age_days, 0) + 1
608
609    result = []
610    for d in range(days - 1, -1, -1):
611        day_date = today - timedelta(days=d)
612        result.append(
613            {
614                "date": day_date.strftime("%Y-%m-%d"),
615                "n": day_counts.get(d, 0),
616                "is_today": d == 0,
617            }
618        )
619    return result

622def collect_summarizer_progress() -> dict:
623    """Return current summarizer progress from ~/.claude/logs/.
624
625    Returns:
626        Dict with keys: exists (bool), total, processed, written, skipped,

647        "processed": processed,
648        "written": data.get("written", 0),
649        "skipped": data.get("skipped", 0),
650        "errors": data.get("errors", 0),
651        "current": data.get("current", ""),
652        "pct": pct,

656def collect_no_db_summary(vault: Path | None = None) -> dict:
657    """Return a file-walk based note count when DB is absent.
658
659    Args:
660        vault: Optional vault path. Defaults to resolve_vault().
661
662    Returns:
663        Dict with keys: vault_exists (bool), total (int),
664        by_folder (list of {folder, n} sorted by n desc).
665    """
666    vault = vault or resolve_vault()
667    if not vault.exists():
668        return {"vault_exists": False, "total": 0, "by_folder": []}
669
670    # ENH-004: this is the explicit no-DB summary, so it always walks -- but it
671    # goes through all_vault_notes() (which falls through to _walk_vault_notes
672    # when there is no DB) so the count excludes EXCLUDE_DIRS, CLAUDE.md,
673    # TAGS.md, and MANIFEST.md exactly like the DB-backed summary does.
674    counts: dict[str, int] = {}
675    total = 0
676    for md in all_vault_notes(vault):
677        folder = md.parent.name if md.parent != vault else "(root)"

559    Args:
560        conn: Open DB connection, or None for file-walk fallback.
561        days: Number of days to cover.
562        vault: Optional vault path. Defaults to resolve_vault().
563
564    Returns:
565        List of dicts with keys: date (YYYY-MM-DD str), n (int), is_today (bool),
566        ordered from oldest to most-recent (index 0 = days-1 ago).
567    """
568    from datetime import date, timedelta
569
570    # Bucket by local calendar day (not rolling 24h windows): bucket index d
571    # holds notes whose mtime falls on calendar day (today - d), matching the
572    # calendar-date labels below regardless of the time of day the report runs.
573    today = date.today()
574    oldest_day = today - timedelta(days=days - 1)
575    # Local midnight at the start of the oldest bucketed day.
576    cutoff_ts = datetime(oldest_day.year, oldest_day.month, oldest_day.day).timestamp()
577
578    day_counts: dict[int, int] = {i: 0 for i in range(days)}
579
580    def _age_days(mtime: float) -> int:
581        age = (today - date.fromtimestamp(mtime)).days
582        return min(max(age, 0), days - 1)
583
584    if conn is not None:
585        rows = fetch_all(
586            conn,
587            "SELECT mtime FROM note_index WHERE mtime >= ?",
588            (cutoff_ts,),
589        )
590        for row in rows:
591            age_days = _age_days(row["mtime"])
592            day_counts[age_days] = day_counts.get(age_days, 0) + 1
593    else:
594        # ENH-004: route through the shared all_vault_notes() helper so this
595        # no-DB walk gets the same EXCLUDE_DIRS / CLAUDE.md / TAGS.md /
596        # MANIFEST.md / SEC-106 symlink filtering as every other read path,
597        # instead of a raw rglob that silently includes them.
598        vault = vault or resolve_vault()
599        if vault.exists():
600            for md in all_vault_notes(vault):
601                try:
602                    mtime = md.stat().st_mtime
603                except OSError:
604                    continue
605                if mtime < cutoff_ts:
606                    continue
607                age_days = _age_days(mtime)
608                day_counts[age_days] = day_counts.get(age_days, 0) + 1
609
610    result = []
611    for d in range(days - 1, -1, -1):
612        day_date = today - timedelta(days=d)
613        result.append(
614            {
615                "date": day_date.strftime("%Y-%m-%d"),
616                "n": day_counts.get(d, 0),
617                "is_today": d == 0,
618            }
619        )
620    return result
