core.transcript_reader

One byte-bounded transcript tail reader for every runtime (ENH-018).

Every hook and adapter reads session transcripts through read_tail() so the byte bound, the path allowlist, and huge-line handling behave the same for Claude, Codex, Gemini, pi, and omp alike. The failure mode this exists for: a large subagent transcript whose SINGLE JSONL line (a giant tool result) exceeds the whole byte budget — tail-by-lines then returns zero usable records and the session summarizes as "No result" and dead-letters. read_tail never discards such a line: it keeps the record and replaces string fields longer than 64 KiB with a <truncated N bytes> marker, so the surrounding dialogue survives.

Stdlib-only.

 64def truncate_oversized_fields(value: object) -> object:
 65    """Recursively cap long strings inside a parsed record.
 66
 67    Strings longer than :data:`_MAX_FIELD_BYTES` become
 68    ``<truncated N bytes>`` (N = the original length). Dicts and lists are
 69    walked; everything else passes through unchanged.
 70    """
 71    if isinstance(value, str):
 72        if len(value) > _MAX_FIELD_BYTES:
 73            return f"<truncated {len(value)} bytes>"
 74        return value
 75    if isinstance(value, dict):
 76        return {k: truncate_oversized_fields(v) for k, v in value.items()}
 77    if isinstance(value, list):
 78        return [truncate_oversized_fields(v) for v in value]
 79    return value

 82def _scan_back_to_line_start(fh, window_start: int, budget: int) -> int | None:
 83    """Find the byte offset where the line containing *window_start* begins.
 84
 85    Reads backwards in chunks of at most *budget* total looking for a
 86    newline before *window_start*; returns the offset just after it, 0 when
 87    the line starts at the file head, or None when no newline is found
 88    within the budget (the line is longer than we allow ourselves to read).
 89    """
 90    seek_to = max(0, window_start - budget)
 91    fh.seek(seek_to)
 92    prefix = fh.read(window_start - seek_to)
 93    idx = prefix.rfind(b"\n")
 94    if idx == -1:
 95        return 0 if seek_to == 0 else None
 96    return seek_to + idx + 1

 99def read_tail(
100    path: Path,
101    *,
102    tail_lines: int,
103    max_bytes: int,
104    max_line_bytes: int = DEFAULT_MAX_LINE_BYTES,
105    cwd: str | None = None,
106    require_allowed: bool = False,
107) -> TranscriptTail:
108    """Read the last *tail_lines* JSONL records of *path*, byte-bounded.
109
110    Algorithm (SEC-111 shape plus huge-line chunking): seek to
111    ``size - max_bytes``, read the window forward, drop the partial first
112    line (cut mid-record), keep at most *tail_lines* whole lines. A line
113    longer than *max_line_bytes* is kept with its record's long string
114    fields truncated (:func:`truncate_oversized_fields`) instead of being
115    discarded — the "No result" dead-letter class disappears because the
116    records on both sides of a multi-MB line survive.
117
118    Args:
119        path: Transcript JSONL path.
120        tail_lines: Maximum number of trailing lines to retain.
121        max_bytes: Ceiling on the trailing byte window.
122        max_line_bytes: Lines longer than this are counted as oversized and
123            field-truncated (default 256 KiB).
124        cwd: Working directory for project-local allowlist roots
125            (``<cwd>/.pi``, ``<cwd>/.gemini``).
126        require_allowed: When True, raise :class:`TranscriptPathError` if
127            *path* is outside the transcript allowlist (SEC-004/SEC-010).
128            Hook entrypoints validate before calling; the default keeps the
129            reader usable on arbitrary paths (tests, --sessions replays).
130
131    Returns:
132        A :class:`TranscriptTail`. Missing files yield an empty tail.
133    """
134    if require_allowed:
135        # Imported lazily: vault_hooks pulls a wide import graph that this
136        # leaf module should not join at import time.
137        from .vault_hooks import is_allowed_transcript_path
138
139        if not is_allowed_transcript_path(path, cwd=cwd):
140            raise TranscriptPathError(f"transcript path outside allowed roots: {path}")
141
142    tail = TranscriptTail()
143    try:
144        size = path.stat().st_size
145    except OSError:
146        return tail
147
148    window = min(size, max_bytes) if max_bytes and max_bytes > 0 else size
149    read_from = size - window
150    try:
151        with path.open("rb") as fh:
152            if read_from > 0:
153                # Does the window start mid-line? (The byte just before it
154                # is not a newline.) If so, recover the line's start by
155                # scanning back at most max_line_bytes so the record is
156                # KEPT (field-truncated below) instead of dropped as a
157                # partial fragment — the "No result" failure mode.
158                fh.seek(read_from - 1)
159                starts_mid_line = fh.read(1) != b"\n"
160                if starts_mid_line:
161                    line_start = _scan_back_to_line_start(fh, read_from, max_line_bytes)
162                    if line_start is not None:
163                        read_from = line_start
164            fh.seek(read_from)
165            raw = fh.read(size - read_from)
166    except OSError:
167        return TranscriptTail()
168
169    tail.bytes_read = size - read_from
170    tail.truncated = read_from > 0
171
172    text = raw.decode("utf-8", errors="replace")
173    lines = text.split("\n")
174    # A trailing newline yields one empty final element; drop it.
175    if lines and lines[-1] == "":
176        lines.pop()
177    if tail_lines > 0:
178        lines = lines[-tail_lines:]
179
180    for line in lines:
181        if not line.strip():
182            continue
183        oversized = len(line.encode("utf-8", errors="replace")) > max_line_bytes
184        if oversized:
185            tail.oversized_lines += 1
186        try:
187            record = json.loads(line)
188        except (json.JSONDecodeError, ValueError):
189            # Not JSON (or truncated head remnant): keep the raw line for
190            # line-based consumers, skip from records.
191            tail.lines.append(line)
192            continue
193        if not isinstance(record, dict):
194            tail.lines.append(line)
195            continue
196        if oversized:
197            shrunk = truncate_oversized_fields(record)
198            if isinstance(shrunk, dict):
199                tail.lines.append(json.dumps(shrunk, ensure_ascii=False))
200                tail.records.append(shrunk)
201            else:  # unreachable: dicts truncate to dicts
202                tail.lines.append(line)
203        else:
204            tail.lines.append(line)
205            tail.records.append(record)
206    return tail
