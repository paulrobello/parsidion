 108#
 109# Frontmatter parsing warnings are printed to stderr (see below), which hook
 110# scripts swallow -- making them invisible to users.  This in-process
 111# collector lets callers (e.g. update_index.py) surface them via
 112# write_hook_event() so they show up in `vault-stats --hooks N`.
 113
 114_PARSE_WARNINGS_MAX = 200
 115# query_note_index defaults to a 200-row cap (sufficient for its original
 116# context-injection callers). The find_notes_by_* wrappers semantically return
 117# EVERY matching note (the walk does), so they pass this effectively-unbounded
 118# limit to override that cap.
 119_FIND_ALL_LIMIT = 10**9

 123def record_parse_warning(msg: str) -> None:
 124    """Record a frontmatter parse warning for later retrieval via drain_parse_warnings().
 125
 126    Capped at ``_PARSE_WARNINGS_MAX`` entries to bound memory during large
 127    index rebuilds; warnings beyond the cap are silently dropped.
 128    """
 129    if len(_parse_warnings) < _PARSE_WARNINGS_MAX:
 130        _parse_warnings.append(msg)

 132
 133def drain_parse_warnings() -> list[str]:
 134    """Return all recorded parse warnings and clear the collector."""
 135    warnings = list(_parse_warnings)
 136    _parse_warnings.clear()
 137    return warnings

 145def parse_frontmatter(content: str) -> dict[str, Any]:
 146    """Parse YAML frontmatter from markdown content using regex.
 147
 148    **Supported YAML subset** (stdlib-only; not a full YAML 1.2 parser) --
 149    scalar/list tokenization lives in ``core/yaml_lite`` (ENH-024); only
 150    the frontmatter-specific structure (delimiters, block sequences,
 151    multi-line scalars) lives here:
 152
 153    - Scalars: bare strings, single/double-quoted strings, integers, floats,
 154      booleans (``true``/``false``/``yes``/``no``), ``null``/``~``, and
 155      date strings (``YYYY-MM-DD`` kept as strings).
 156    - Inline lists: ``key: [a, b, c]`` with optional quoting of items.
 157      Quoted items may contain commas.  List items are always kept as
 158      strings (never coerced to bool/int/float) so numeric-looking tags
 159      like ``tags: [2026, python]`` remain findable.
 160    - Block sequence lists: ``key:`` followed by ``  - item`` lines
 161      (items kept as strings, same as inline lists).
 162    - Multi-line scalars: ``>`` (folded -- joins continuation lines with a
 163      space), ``|`` (literal -- joins with newlines), and strip variants
 164      ``>-`` / ``|-``.  Only indented continuation lines (indent > 0) are
 165      collected; the block ends at the next bare key or blank line.
 166    - Comment LINES (first non-blank character ``#``) are skipped; trailing
 167      inline comments are NOT stripped from scalar values (``key: v # c``
 168      reads back as the string ``v # c``), unlike the config.yaml parser.
 169
 170    **Not supported** (silently ignored or returned as bare strings):
 171    - Nested mappings deeper than 1 level (``key: {a: 1}`` or indented
 172      sub-mappings).
 173    - YAML anchors, aliases, and tags (``!!str``, etc.).
 174    - Multi-document streams (``---`` as a separator within a value).
 175    - Flow mappings.
 176
 177    Returns an empty dict when no frontmatter block is found or when the
 178    opening/closing ``---`` delimiters are missing.
 179    """
 180    match = _FRONTMATTER_RE.match(content)
 181    if not match:
 182        return {}
 183
 184    raw = match.group(1)
 185    result: dict[str, Any] = {}
 186    current_key: str | None = None
 187    current_list: list[str] | None = None
 188    # Multi-line scalar state: block_style is ">" (folded) or "|" (literal)
 189    block_style: str | None = None
 190    block_parts: list[str] = []
 191
 192    def _flush_block() -> None:
 193        """Finalize a multi-line scalar block and store it in result."""
 194        nonlocal block_style, block_parts
 195        if current_key is not None and block_style is not None and block_parts:
 196            if block_style in (">", ">-"):
 197                result[current_key] = " ".join(block_parts)
 198            else:  # "|" or "|-"
 199                result[current_key] = "\n".join(block_parts)
 200        block_style = None
 201        block_parts = []
 202
 203    for line in raw.splitlines():
 204        stripped = line.strip()
 205        indent = len(line) - len(line.lstrip())
 206
 207        # If we're collecting a multi-line scalar block, check for continuation
 208        if block_style is not None:
 209            if indent > 0 and stripped:
 210                block_parts.append(stripped)
 211                continue
 212            else:
 213                _flush_block()
 214                # Fall through to process this line normally
 215
 216        # Continuation of a multi-line list (- item form)
 217        if (
 218            stripped.startswith("- ")
 219            and current_key is not None
 220            and current_list is not None
 221        ):
 222            current_list.append(parse_list_item(stripped[2:].strip()))
 223            result[current_key] = current_list
 224            continue
 225
 226        # If we were collecting a list and hit a non-list line, close it
 227        if current_list is not None:
 228            current_list = None
 229
 230        # Skip blank lines and comments
 231        if not stripped or stripped.startswith("#"):
 232            continue
 233
 234        # key: value pair
 235        key_value = split_key_value(line)
 236        if key_value is None:
 237            continue
 238
 239        key, value_str = key_value
 240
 241        if not key:
 242            continue
 243
 244        # QA-008: warn when an indented sub-key looks like a nested mapping.
 245        # Nested mappings are not supported by this parser (see docstring).
 246        # Emit a stderr warning so vault-doctor checks don't silently mislead.
 247        # This is safe: hook scripts print JSON to stdout; stderr is ignored.
 248        if indent > 0 and key:
 249            _warning = (
 250                f"parse_frontmatter: nested mapping key '{key}' (indented) is not "
 251                "supported and will be treated as a top-level scalar. "
 252                "Consider flattening this YAML key."
 253            )
 254            print(_warning, file=sys.stderr)
 255            record_parse_warning(_warning)
 256
 257        current_key = key
 258
 259        # Empty value -- could be start of a multi-line list
 260        if not value_str:
 261            current_list = []
 262            result[key] = current_list
 263            continue
 264
 265        # Multi-line scalar block indicators: >, |, >-, |-
 266        if value_str in (">", "|", ">-", "|-"):
 267            block_style = value_str
 268            block_parts = []
 269            current_list = None
 270            continue
 271
 272        # Inline list: [a, b, c]
 273        items = parse_inline_list(value_str)
 274        if items is not None:
 275            result[key] = items
 276            current_list = None
 277            continue
 278
 279        # Scalar value
 280        result[key] = parse_scalar(value_str)
 281        current_list = None
 282
 283    # Flush any remaining block at end of frontmatter
 284    _flush_block()
 285
 286    return result

 293# Scalar/array emission policy (quoting, round-trip rules) lives in
 294# core/yaml_lite (ENH-024): scalar_needs_quotes / quote_scalar /
 295# dump_scalar / dump_list_item. Only the frontmatter-specific policy
 296# stays here.
 297# List fields whose items are always double-quoted: ``related`` holds
 298# ``[[wikilinks]]`` and the canonical form (CLAUDE.md conventions,
 299# visualizer/lib/frontmatter.ts) is ``related: ["[[a]]", "[[b]]"]``.
 300_ALWAYS_QUOTED_LIST_KEYS: frozenset[str] = frozenset({"related"})
 301
 302
 303def serialize_frontmatter(fields: dict[str, Any]) -> str:
 304    """Serialize a frontmatter dict to the canonical YAML block.
 305
 306    ARC-005: the single schema-aware emitter shared by ``vault_new``,
 307    ``vault_merge``, the ``tools/migrate_*`` importers, and (via the parity
 308    fixture ``tests/fixtures/parity/frontmatter.json``) the visualizer's
 309    ``frontmatter.ts``. Replaces four hand-built ``_build_frontmatter``
 310    copies whose quoting and key order had drifted.
 311
 312    Canonical form:
 313
 314    - keys in :data:`note_schema.FRONTMATTER_FIELD_ORDER` first (when
 315      present), remaining keys after them in insertion order;
 316    - ``None`` and empty-string values are dropped (the writer that has an
 317      opinion about empties filters before calling);
 318    - ``tags``/``sources`` as inline arrays with bare items unless an item
 319      needs quoting; ``related`` items always double-quoted (the
 320      ``["[[wikilink]]"]`` convention);
 321    - scalars bare unless quoting is required for an exact round-trip
 322      through :func:`parse_frontmatter`.
 323
 324    Returns the ``---\\n...\\n---\\n`` block (no trailing blank line —
 325    callers append the body).
 326    """
 327    ordered = [k for k in FRONTMATTER_FIELD_ORDER if k in fields]
 328    ordered += [k for k in fields if k not in FRONTMATTER_FIELD_ORDER]
 329
 330    lines: list[str] = ["---"]
 331    for key in ordered:
 332        value = fields[key]
 333        if value is None or value == "":
 334            continue
 335        if isinstance(value, list):
 336            if value:
 337                items = ", ".join(
 338                    dump_list_item(item, always_quote=key in _ALWAYS_QUOTED_LIST_KEYS)
 339                    for item in value
 340                )
 341                lines.append(f"{key}: [{items}]")
 342            else:
 343                lines.append(f"{key}: []")
 344        else:
 345            lines.append(f"{key}: {dump_scalar(value)}")
 346    lines.append("---")
 347    return "\n".join(lines) + "\n"

 350def get_body(content: str) -> str:
 351    """Return markdown content after the frontmatter block.
 352
 353    If no frontmatter is found, returns the entire content.
 354    """
 355    match = _FRONTMATTER_RE.match(content)
 356    if not match:
 357        return content
 358    return content[match.end() :]

 361def extract_title(content: str, stem: str) -> str:
 362    """Extract the display title from a vault note.
 363
 364    Searches the note body for the first top-level ``# `` heading (a single
 365    hash followed by a space, never ``##`` or deeper).  Falls back to the
 366    filename *stem* converted to title-case if no heading is found.
 367
 368    This is the canonical title-extraction function for the vault.  All
 369    scripts that need a note title should call this instead of duplicating
 370    the logic.  See ARC-009.
 371
 372    Args:
 373        content: Full note content (frontmatter + body).
 374        stem: Filename stem (without extension) used as fallback title.
 375
 376    Returns:
 377        Title string -- either the heading text or the humanized stem.
 378    """
 379    body = get_body(content)
 380    for line in body.splitlines():
 381        stripped = line.strip()
 382        if stripped.startswith("# ") and not stripped.startswith("## "):
 383            return stripped[2:].strip()
 384    return stem.replace("-", " ").title()

 392def slugify(text: str) -> str:
 393    """Convert text to a kebab-case filename slug.
 394
 395    Lowercases the text, replaces spaces and underscores with hyphens,
 396    transliterates non-ASCII characters to their closest ASCII equivalent
 397    (e.g. "é" -> "e"), removes remaining special characters, and collapses
 398    multiple consecutive hyphens.
 399
 400    Titles that are entirely non-transliterable (e.g. CJK) would otherwise
 401    collapse to an empty slug and collide with each other; in that case a
 402    stable hash of the original title is used instead so distinct titles
 403    never collide. Purely-ASCII titles that strip to nothing (e.g. "!!!")
 404    still return "" -- vault_new.py relies on that to reject garbage titles.
 405    """
 406    has_non_ascii = any(ord(ch) > 127 for ch in text)
 407    slug = text.lower().strip()
 408    slug = slug.replace(" ", "-").replace("_", "-")
 409    # Transliterate accented/non-ASCII characters before stripping specials
 410    # so e.g. "café" -> "cafe" instead of "caf".
 411    slug = unicodedata.normalize("NFKD", slug).encode("ascii", "ignore").decode()
 412    slug = _SLUG_SPECIAL_RE.sub("", slug)
 413    slug = _SLUG_MULTI_HYPHEN_RE.sub("-", slug)
 414    slug = slug.strip("-")
 415    if not slug and has_non_ascii:
 416        slug = "note-" + hashlib.sha1(text.encode()).hexdigest()[:8]
 417    return slug

 425def ensure_note_index_schema(conn: sqlite3.Connection) -> None:
 426    """Create the note_index table and indexes if they don't exist.
 427
 428    Also idempotently adds the ``date`` column to databases created before
 429    Phase 2 of the memory-enhancements plan (point-in-time search support).
 430
 431    Args:
 432        conn: An open sqlite3.Connection (caller sets WAL mode).
 433    """
 434    conn.execute(
 435        """
 436        CREATE TABLE IF NOT EXISTS note_index (
 437            stem           TEXT    NOT NULL PRIMARY KEY,
 438            path           TEXT    NOT NULL,
 439            folder         TEXT    NOT NULL DEFAULT '',
 440            title          TEXT    NOT NULL DEFAULT '',
 441            summary        TEXT    NOT NULL DEFAULT '',
 442            tags           TEXT    NOT NULL DEFAULT '',
 443            note_type      TEXT    NOT NULL DEFAULT '',
 444            project        TEXT    NOT NULL DEFAULT '',
 445            confidence     TEXT    NOT NULL DEFAULT '',
 446            mtime          REAL    NOT NULL DEFAULT 0.0,
 447            related        TEXT    NOT NULL DEFAULT '',
 448            is_stale       INTEGER NOT NULL DEFAULT 0,
 449            incoming_links INTEGER NOT NULL DEFAULT 0,
 450            date           TEXT    NOT NULL DEFAULT '',
 451            prompt_version TEXT    NOT NULL DEFAULT '',
 452            incoming_stems TEXT    NOT NULL DEFAULT ''
 453        )
 454        """
 455    )
 456    # Migration: add date column to pre-existing databases (CREATE IF NOT EXISTS
 457    # does not add columns to an already-present table).
 458    cols = {row[1] for row in conn.execute("PRAGMA table_info(note_index)")}
 459    if "date" not in cols:
 460        conn.execute("ALTER TABLE note_index ADD COLUMN date TEXT NOT NULL DEFAULT ''")
 461    # ENH-008 Step 3: prompt_version column for slicing note quality by the
 462    # prompt that produced each AI-generated note.
 463    if "prompt_version" not in cols:
 464        conn.execute(
 465            "ALTER TABLE note_index ADD COLUMN prompt_version TEXT NOT NULL DEFAULT ''"
 466        )
 467    # ENH-021: incoming_stems column -- the reverse-link adjacency (JSON array
 468    # of source stems), inverted from every note's outgoing ``related`` links
 469    # at index time. Empty on pre-ENH-021 rows until the next full rebuild;
 470    # readers treat empty as "not populated" and fall back to deriving the
 471    # inversion from ``related`` (see SessionIndexSnapshot).
 472    if "incoming_stems" not in cols:
 473        conn.execute(
 474            "ALTER TABLE note_index ADD COLUMN incoming_stems TEXT NOT NULL DEFAULT ''"
 475        )
 476    conn.execute("CREATE INDEX IF NOT EXISTS idx_ni_folder    ON note_index(folder)")
 477    conn.execute("CREATE INDEX IF NOT EXISTS idx_ni_note_type ON note_index(note_type)")
 478    conn.execute("CREATE INDEX IF NOT EXISTS idx_ni_project   ON note_index(project)")
 479    conn.execute(
 480        "CREATE INDEX IF NOT EXISTS idx_ni_mtime     ON note_index(mtime DESC)"
 481    )
 482    conn.execute("CREATE INDEX IF NOT EXISTS idx_ni_tags      ON note_index(tags)")
 483    conn.execute("CREATE INDEX IF NOT EXISTS idx_ni_date      ON note_index(date)")
 484    conn.commit()

 487def _paths_from_rows(rows: list[Any], vault_root_resolved: Path) -> list[Path]:
 488    """Re-validate DB-derived path strings against the vault root.
 489
 490    SEC-005 / SEC-130: a tampered ``embeddings.db`` can store any path string;
 491    never trust SQLite output. Each path must exist and resolve inside the
 492    vault. Follows the precedent established inline in ``query_note_index``
 493    (the original vault_index.py:373-420 containment guard).
 494
 495    Args:
 496        rows: Rows from a ``SELECT path ...`` query. Each row is either a
 497            ``(path_str,)`` tuple/list or a bare string.
 498        vault_root_resolved: The resolved vault root to check containment against.
 499
 500    Returns:
 501        List of existing, in-vault ``Path`` objects (unsorted -- caller's SQL
 502        determines order).
 503    """
 504    result: list[Path] = []
 505    for row in rows:
 506        path_str = row[0] if isinstance(row, (tuple, list)) else str(row)
 507        p = Path(path_str)
 508        if p.exists() and is_path_inside_vault(p, vault_root_resolved):
 509            result.append(p)
 510    return result

 527def _build_note_index_where(
 528    *,
 529    tag: str | None = None,
 530    folder: str | None = None,
 531    note_type: str | None = None,
 532    project: str | None = None,
 533    recent_days: int | None = None,
 534    changed_since: str | None = None,
 535    as_of: str | None = None,
 536) -> tuple[str, list[object]]:
 537    """Build the note_index metadata-filter WHERE clause (QA-009).
 538
 539    The single home for the condition assembly previously duplicated
 540    between ``query_note_index`` here and ``cli.search.metadata.query``
 541    (parsight similarity 0.908). Callers keep their own SELECT lists,
 542    result mapping, and schema/containment guards.
 543
 544    SECURITY: The SQL WHERE clause is assembled from literal condition
 545    fragments only -- no column names are ever derived from external input.
 546    All filter values are passed as bound parameters (?). Column names used
 547    form a static whitelist: tags, folder, note_type, project, mtime, date.
 548    Any future addition of a user-supplied column name must be added to
 549    this whitelist and reviewed for injection risk.
 550
 551    Args:
 552        tag: Exact tag token to match in the comma-separated tags column.
 553        folder: Exact folder name to match.
 554        note_type: Exact note_type value to match.
 555        project: Exact project value to match.
 556        recent_days: Only return notes modified within this many days.
 557        changed_since: Only return notes modified on/after this ISO datetime.
 558        as_of: Only return notes whose frontmatter date is on/before this
 559            ISO date (empty dates excluded).
 560
 561    Returns:
 562        ``(where, params)`` -- ``where`` is "" or "WHERE c1 AND c2 ...";
 563        ``params`` holds the bound values in placeholder order.
 564    """
 565    conditions: list[str] = []
 566    params: list[object] = []
 567
 568    if tag is not None:
 569        # 4-pattern exact-token match to avoid partial hits (e.g. "python"
 570        # must not match "python-decorator").  Tags are stored as
 571        # ", ".join(sorted(tags_list)) -- canonical format enforced at write
 572        # time in update_index.py and build_embeddings.py.  See ARC-004.
 573        conditions.append("(tags = ? OR tags LIKE ? OR tags LIKE ? OR tags LIKE ?)")
 574        params.extend([tag, f"{tag},%", f"%, {tag}", f"%, {tag},%"])
 575
 576    if folder is not None:
 577        conditions.append("folder = ?")
 578        params.append(folder)
 579
 580    if note_type is not None:
 581        conditions.append("note_type = ?")
 582        params.append(note_type)
 583
 584    if project is not None:
 585        conditions.append("project = ?")
 586        params.append(project)
 587
 588    if recent_days is not None:
 589        cutoff = (datetime.now() - timedelta(days=recent_days)).timestamp()
 590        conditions.append("mtime >= ?")
 591        params.append(cutoff)
 592
 593    if changed_since is not None:
 594        cutoff = datetime.fromisoformat(changed_since).timestamp()
 595        conditions.append("mtime >= ?")
 596        params.append(cutoff)
 597
 598    if as_of is not None:
 599        # ISO YYYY-MM-DD strings sort lexicographically; exclude empty dates.
 600        conditions.append("date != '' AND date <= ?")
 601        params.append(as_of)
 602
 603    where = ("WHERE " + " AND ".join(conditions)) if conditions else ""
 604    return where, params

 607def query_note_index(
 608    *,
 609    tag: str | None = None,
 610    folder: str | None = None,
 611    note_type: str | None = None,
 612    project: str | None = None,
 613    recent_days: int | None = None,
 614    limit: int = 200,
 615    vault: str | Path | None = None,
 616) -> list[Path] | None:
 617    """Query the note_index table in embeddings.db for fast metadata filtering.
 618
 619    Returns None (not []) if the DB is absent or the table is missing,
 620    signalling the caller to fall back to a file walk. An empty result set
 621    is returned as ``[]`` and is distinguishable from ``None`` -- this is what
 622    makes the DB-first ``find_notes_by_*`` fallback correct (a project with
 623    zero notes returns ``[]``, not the walk fallback).
 624
 625    Args:
 626        tag: Exact tag token to match in the comma-separated tags column.
 627        folder: Exact folder name to match.
 628        note_type: Exact note_type value to match.
 629        project: Exact project value to match.
 630        recent_days: Only return notes modified within this many days.
 631        limit: Maximum number of results (default 200).
 632        vault: Optional vault path used to locate embeddings.db and to validate
 633            returned paths against. Defaults to resolve_vault().
 634
 635    Returns:
 636        List of existing Paths sorted by mtime descending, or None on DB error.
 637    """
 638    if isinstance(vault, str):  # be liberal: accept str paths from callers
 639        vault = Path(vault)
 640    db_path = get_embeddings_db_path(vault)
 641    if not db_path.exists():
 642        return None
 643
 644    try:
 645        conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
 646    except sqlite3.OperationalError:
 647        return None
 648
 649    try:
 650        row = conn.execute(
 651            "SELECT 1 FROM sqlite_master WHERE type='table' AND name='note_index'"
 652        ).fetchone()
 653        if row is None:
 654            return None
 655
 656        # QA-009: shared WHERE builder (see _build_note_index_where for the
 657        # injection-safety contract).
 658        where, params = _build_note_index_where(
 659            tag=tag,
 660            folder=folder,
 661            note_type=note_type,
 662            project=project,
 663            recent_days=recent_days,
 664        )
 665        sql = f"SELECT path FROM note_index {where} ORDER BY mtime DESC LIMIT ?"
 666        params.append(limit)
 667
 668        rows = conn.execute(sql, params).fetchall()
 669        # SEC-005 / SEC-130: paths from SQLite are re-validated against the
 670        # vault root via _paths_from_rows -- a tampered embeddings.db cannot
 671        # inject reads outside the vault.
 672        vault_root_resolved = (vault or resolve_vault()).resolve()
 673        return _paths_from_rows(rows, vault_root_resolved)
 674    except sqlite3.Error:
 675        return None
 676    finally:
 677        conn.close()

 685def _walk_vault_notes(vault: str | Path | None = None) -> list[Path]:
 686    """Walk the vault tree and return all .md files, excluding EXCLUDE_DIRS, CLAUDE.md, TAGS.md, and MANIFEST.md.
 687
 688    Args:
 689        vault: Optional vault path. Defaults to resolve_vault().
 690    """
 691    if isinstance(vault, str):  # be liberal: accept str paths from callers
 692        vault = Path(vault)
 693    vault = vault or resolve_vault()
 694    notes: list[Path] = []
 695    if not vault.is_dir():
 696        return notes
 697
 698    # SEC-106: resolve once so the symlink guard is not defeated by a
 699    # symlinked vault root. ``Templates/`` is excluded via EXCLUDE_DIRS
 700    # *before* the symlink guard runs, so the intentional
 701    # ``Templates -> skills/parsidion/templates`` symlink is preserved.
 702    vault_root_resolved = vault.resolve()
 703
 704    for dirpath, dirnames, filenames in os.walk(vault):
 705        # Prune excluded directories in-place so os.walk skips them
 706        dirnames[:] = [d for d in dirnames if d not in EXCLUDE_DIRS]
 707
 708        for fname in filenames:
 709            if not fname.endswith(".md"):
 710                continue
 711            if Path(dirpath) == vault and fname in ("CLAUDE.md", "TAGS.md"):
 712                continue
 713            if fname == "MANIFEST.md":
 714                continue
 715            p = Path(dirpath) / fname
 716            # SEC-106: skip symlinked .md files whose target escapes the
 717            # vault; ``os.walk`` lists them like regular files.
 718            if not is_symlink_inside_vault(p, vault_root_resolved):
 719                continue
 720            notes.append(p)
 721
 722    return notes

 725def _find_notes_by_field(
 726    field: str, value: str, vault: str | Path | None = None
 727) -> list[Path]:
 728    """Find all notes where a frontmatter *field* matches *value* (case-insensitive).
 729
 730    For scalar fields (``project``, ``type``), matches the value directly.
 731    For list fields (``tags``), matches if any element equals *value*.
 732
 733    Args:
 734        field: The frontmatter field name to search (e.g. ``"project"``).
 735        value: The target value to match (compared case-insensitively).
 736        vault: Optional vault path. Defaults to resolve_vault().
 737
 738    Returns:
 739        List of matching note paths.
 740    """
 741    matches: list[Path] = []
 742    target = value.lower()
 743    for note_path in _walk_vault_notes(vault):
 744        try:
 745            content = note_path.read_text(encoding="utf-8")

 747            continue

 753            if any(
 754                isinstance(item, str) and item.lower() == target for item in field_val
 755            ):
 756                matches.append(note_path)
 757    return matches

 890def read_note_summary(path: Path, max_lines: int = 5) -> str:
 891    """Read a note and return its title (first ``#`` heading) plus the first
 892    *max_lines* of body content. Used for building context blocks.
 893    """
 894    try:
 895        content = path.read_text(encoding="utf-8")
 896    except (OSError, UnicodeDecodeError):
 897        return ""
 898
 899    body = get_body(content)
 900    lines = body.strip().splitlines()
 901
 902    title: str = path.stem  # fallback title
 903    body_start = 0
 904
 905    for i, line in enumerate(lines):
 906        stripped = line.strip()
 907        if stripped.startswith("#"):
 908            title = stripped.lstrip("#").strip()
 909            body_start = i + 1
 910            break
 911
 912    # Collect up to max_lines of non-empty body content after the title
 913    summary_lines: list[str] = []
 914    for line in lines[body_start:]:
 915        stripped = line.strip()
 916        if not stripped:
 917            continue
 918        # Skip HTML comments
 919        if stripped.startswith("<!--"):
 920            continue
 921        summary_lines.append(stripped)
 922        if len(summary_lines) >= max_lines:
 923            break
 924
 925    result = title
 926    if summary_lines:
 927        result += "\n" + "\n".join(summary_lines)
 928    return result

 971def note_index_age(vault: str | Path | None = None) -> float:
 972    """Return how stale ``note_index`` is, in seconds.
 973
 974    Computes ``max(on-disk .md mtime) - max(note_index mtime)``. Returns
 975    ``0.0`` when the index is current or ahead, when the DB is absent, or
 976    when either source is unreadable. A positive value means on-disk notes
 977    exist that the index does not yet reflect -- the DB-first read path may
 978    return stale results, and the user should rebuild via
 979    ``update_index.py``.
 980
 981    This is a *signal*, not a trigger: read paths never auto-rebuild, because
 982    a read doing surprising write work is worse than a slightly stale result.
 983
 984    Args:
 985        vault: Optional vault path. Defaults to resolve_vault().
 986    """
 987    resolved = vault or resolve_vault()
 988    db_path = get_embeddings_db_path(resolved)
 989    if not db_path.exists():
 990        return 0.0
 991    try:
 992        conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
 993    except sqlite3.OperationalError:
 994        return 0.0
 995    try:
 996        row = conn.execute("SELECT MAX(mtime) FROM note_index").fetchone()
 997    except sqlite3.Error:
 998        conn.close()
 999        return 0.0
1000    conn.close()
1001    db_max = float(row[0]) if row and row[0] is not None else 0.0
1002
1003    disk_max = 0.0
1004    for p in _walk_vault_notes(resolved):
1005        try:
1006            m = p.stat().st_mtime
1007        except OSError:
1008            continue
1009        if m > disk_max:
1010            disk_max = m
1011    if disk_max <= db_max:
1012        return 0.0
1013    return disk_max - db_max

1021def build_context_block(notes: list[Path], max_chars: int = 4000) -> str:
1022    """Build a context string from a list of notes, respecting *max_chars* budget.
1023
1024    Each note is formatted as::
1025
1026        ### Note Title (folder/filename)
1027        [summary lines]
1028
1029    Stops adding notes when approaching *max_chars*.
1030    """
1031    parts: list[str] = []
1032    char_count = 0
1033
1034    vault_root = resolve_vault()
1035    for note_path in notes:
1036        # Determine the relative folder/filename label
1037        try:
1038            rel = note_path.relative_to(vault_root)
1039        except ValueError:
1040            rel = Path(note_path.parent.name) / note_path.name
1041
1042        summary = read_note_summary(note_path)
1043        if not summary:
1044            continue
1045
1046        # Extract title from first line of summary
1047        summary_lines = summary.splitlines()
1048        title = summary_lines[0] if summary_lines else note_path.stem
1049        body = "\n".join(summary_lines[1:]) if len(summary_lines) > 1 else ""
1050
1051        block = f"### {title} ({rel})\n"
1052        if body:
1053            block += body + "\n"
1054        block += "\n"
1055
1056        if char_count + len(block) > max_chars:
1057            break
1058
1059        parts.append(block)
1060        char_count += len(block)
1061
1062    return "".join(parts).rstrip("\n")

1105class SessionIndexSnapshot:
1106    """One read of ``note_index``, shared across a single hook invocation.
1107
1108    PRF-104: a standard SessionStart used to perform roughly three full
1109    ``note_index`` reads (the project and recent queries, the graph-metadata
1110    load, and another full load inside ``build_compact_index``), each opening
1111    its own connection, plus an O(N x seeds) incoming-link scan rebuilt per
1112    call. None of that is algorithmically wrong -- it is the SHAPE of the work
1113    that grows with the vault -- so the fix is one snapshot per run rather than
1114    micro-optimising each site.
1115
1116    The derived views (compact-index map, graph metadata, and the reverse-link
1117    adjacency the Tier-1 neighbour expansion needs) are computed once here in a
1118    single pass. Path validation is NOT skipped: callers that resolve rows to
1119    filesystem paths still apply the SEC-005 / SEC-130 containment guard -- the
1120    saving is doing it once per run instead of once per query.
1121
1122    Construct via :func:`load_session_index_snapshot`, which returns ``None``
1123    when the DB or table is absent, mirroring ``query_note_index``'s sentinel
1124    contract so every caller keeps its existing fallback.
1125    """
1126
1127    __slots__ = ("vault", "rows", "by_stem", "_graph_meta", "_incoming", "_compact")
1128
1129    def __init__(self, vault: Path, rows: list[SessionIndexRow]) -> None:
1130        self.vault = vault
1131        self.rows = rows
1132        self.by_stem: dict[str, SessionIndexRow] = {r.stem: r for r in rows}
1133
1134        self._graph_meta: dict[str, dict[str, object]] = {
1135            r.stem: {
1136                "path": r.path,
1137                "related": r.related,
1138                "incoming_links": r.incoming_links,
1139                "tags": r.tags,
1140            }
1141            for r in rows
1142        }
1143        self._compact: dict[str, tuple[str, str, str]] = {
1144            r.stem: (r.title, r.tags, r.folder) for r in rows
1145        }
1146        # Reverse adjacency. ENH-021: read the inversion persisted at index
1147        # time in the ``incoming_stems`` column; only when the column is not
1148        # populated (pre-ENH-021 index) rebuild it here in one
1149        # O(N x avg_links) pass -- the PRF-104 behaviour.
1150        incoming = _incoming_stems_from_column(rows)
1151        if incoming is None:
1152            incoming = {}
1153            for row in rows:
1154                for target in parse_related_stems(row.related):
1155                    incoming.setdefault(target, set()).add(row.stem)
1156        self._incoming = incoming
1157
1158    def graph_metadata(self) -> dict[str, dict[str, object]]:
1159        """The mapping ``load_graph_metadata`` returns, computed once."""
1160        return self._graph_meta
1161
1162    def compact_index_map(self) -> dict[str, tuple[str, str, str]]:
1163        """The stem -> (title, tags, folder) map ``build_compact_index`` uses."""
1164        return self._compact
1165
1166    def incoming_stems(self, stem: str) -> set[str]:
1167        """Stems whose ``related`` field points at *stem* (precomputed)."""
1168        return self._incoming.get(stem, set())

1170    def paths_where(
1171        self,
1172        *,
1173        project: str | None = None,
1174        recent_days: int | None = None,
1175        limit: int = 200,
1176    ) -> list[Path]:
1177        """Rows matching the filters, newest first, as validated Paths.
1178
1179        Mirrors ``query_note_index``'s filtering, ordering, containment guard
1180        and limit semantics -- against the snapshot instead of a fresh query.
1181        """
1182        cutoff = (
1183            (datetime.now() - timedelta(days=recent_days)).timestamp()
1184            if recent_days is not None
1185            else None

150    scalar/list tokenization lives in ``core/yaml_lite`` (ENH-024); only

165      ``>-`` / ``|-``.  Only indented continuation lines (indent > 0) are

169      reads back as the string ``v # c``), unlike the config.yaml parser.

172    - Nested mappings deeper than 1 level (``key: {a: 1}`` or indented
173      sub-mappings).
174    - YAML anchors, aliases, and tags (``!!str``, etc.).
175    - Multi-document streams (``---`` as a separator within a value).
176    - Flow mappings.

179    opening/closing ``---`` delimiters are missing.

221            and current_list is not None

394    """Convert text to a kebab-case filename slug.
395
396    Lowercases the text, replaces spaces and underscores with hyphens,
397    transliterates non-ASCII characters to their closest ASCII equivalent
398    (e.g. "é" -> "e"), removes remaining special characters, and collapses
399    multiple consecutive hyphens.
400
401    Titles that are entirely non-transliterable (e.g. CJK) would otherwise
402    collapse to an empty slug and collide with each other; in that case a
403    stable hash of the original title is used instead so distinct titles
404    never collide. Purely-ASCII titles that strip to nothing (e.g. "!!!")
405    still return "" -- vault_new.py relies on that to reject garbage titles.
406    """
407    has_non_ascii = any(ord(ch) > 127 for ch in text)
408    slug = text.lower().strip()
409    slug = slug.replace(" ", "-").replace("_", "-")
410    # Transliterate accented/non-ASCII characters before stripping specials
411    # so e.g. "café" -> "cafe" instead of "caf".
412    slug = unicodedata.normalize("NFKD", slug).encode("ascii", "ignore").decode()
413    slug = _SLUG_SPECIAL_RE.sub("", slug)
414    slug = _SLUG_MULTI_HYPHEN_RE.sub("-", slug)
415    slug = slug.strip("-")
416    if not slug and has_non_ascii:
417        slug = "note-" + hashlib.sha1(text.encode()).hexdigest()[:8]
418    return slug

686def _walk_vault_notes(vault: str | Path | None = None) -> list[Path]:
687    """Walk the vault tree and return all .md files, excluding EXCLUDE_DIRS, CLAUDE.md, TAGS.md, and MANIFEST.md.
688
689    Args:
690        vault: Optional vault path. Defaults to resolve_vault().
691    """
692    if isinstance(vault, str):  # be liberal: accept str paths from callers
693        vault = Path(vault)
694    vault = vault or resolve_vault()
695    notes: list[Path] = []
696    if not vault.is_dir():
697        return notes
698
699    # SEC-106: resolve once so the symlink guard is not defeated by a
700    # symlinked vault root. ``Templates/`` is excluded via EXCLUDE_DIRS
701    # *before* the symlink guard runs, so the intentional
702    # ``Templates -> skills/parsidion/templates`` symlink is preserved.
703    vault_root_resolved = vault.resolve()
704
705    for dirpath, dirnames, filenames in os.walk(vault):
706        # Prune excluded directories in-place so os.walk skips them
707        dirnames[:] = [d for d in dirnames if d not in EXCLUDE_DIRS]
708
709        for fname in filenames:
710            if not fname.endswith(".md"):
711                continue
712            if Path(dirpath) == vault and fname in ("CLAUDE.md", "TAGS.md"):
713                continue
714            if fname == "MANIFEST.md":
715                continue
716            p = Path(dirpath) / fname
717            # SEC-106: skip symlinked .md files whose target escapes the
718            # vault; ``os.walk`` lists them like regular files.
719            if not is_symlink_inside_vault(p, vault_root_resolved):
720                continue
721            notes.append(p)
722
723    return notes
