vault_common

Shared utility library for the Parsidion vault knowledge management system.

Provides functions for parsing frontmatter, searching notes, building context blocks, and managing the vault directory structure. Uses only Python stdlib.

ARC-005: This module is now a thin re-export facade. The implementation has been split into focused sub-modules:

  • vault_config -- config loading, parsing, validation
  • vault_path -- path resolution, vault constants, secure logging
  • vault_fs -- file locking, pending queue, git, daily notes
  • vault_index -- frontmatter parsing, note search, context building
  • vault_hooks -- hook event logging, env helpers, transcript analysis
  • vault_adaptive -- per-note usefulness tracking, last-seen state

The supported public API is re-exported and enumerated in __all__, so existing callers (import vault_common; vault_common.X()) keep working unchanged. A handful of underscore-prefixed private symbols are ALSO re-exported purely for backward-compat test access — they are importable from vault_common but deliberately omitted from __all__; new code should import them from their defining submodule instead.

120def rotate_log_file(log_path: Path, max_lines: int = _HOOK_ERROR_LOG_MAX_LINES) -> None:
121    """Rotate a log file when it exceeds *max_lines*, keeping the second half.
122
123    Best-effort -- never raises.
124
125    Args:
126        log_path: Path to the log file to rotate.
127        max_lines: Maximum number of lines before rotation is triggered.
128    """
129    try:
130        if not log_path.exists():
131            return
132        lines = log_path.read_text(encoding="utf-8").splitlines(keepends=True)
133        if len(lines) <= max_lines:
134            return
135        keep = lines[max_lines // 2 :]
136        log_path.write_text("".join(keep), encoding="utf-8")
137    except OSError:
138        pass

134def parse_frontmatter(content: str) -> dict[str, Any]:
135    """Parse YAML frontmatter from markdown content using regex.
136
137    **Supported YAML subset** (stdlib-only; not a full YAML 1.2 parser):
138
139    - Scalars: bare strings, single/double-quoted strings, integers, floats,
140      booleans (``true``/``false``/``yes``/``no``), ``null``/``~``, and
141      date strings (``YYYY-MM-DD`` kept as strings).
142    - Inline lists: ``key: [a, b, c]`` with optional quoting of items.
143      Quoted items may contain commas.  List items are always kept as
144      strings (never coerced to bool/int/float) so numeric-looking tags
145      like ``tags: [2026, python]`` remain findable.
146    - Block sequence lists: ``key:`` followed by ``  - item`` lines
147      (items kept as strings, same as inline lists).
148    - Multi-line scalars: ``>`` (folded -- joins continuation lines with a
149      space), ``|`` (literal -- joins with newlines), and strip variants
150      ``>-`` / ``|-``.  Only indented continuation lines (indent > 0) are
151      collected; the block ends at the next bare key or blank line.
152    - Trailing inline comments (``# comment``) are stripped from scalar
153      values, respecting surrounding quotes.
154
155    **Not supported** (silently ignored or returned as bare strings):
156    - Nested mappings deeper than 1 level (``key: {a: 1}`` or indented
157      sub-mappings).
158    - YAML anchors, aliases, and tags (``!!str``, etc.).
159    - Multi-document streams (``---`` as a separator within a value).
160    - Flow mappings.
161
162    Returns an empty dict when no frontmatter block is found or when the
163    opening/closing ``---`` delimiters are missing.
164    """
165    match = _FRONTMATTER_RE.match(content)
166    if not match:
167        return {}
168
169    raw = match.group(1)
170    result: dict[str, Any] = {}
171    current_key: str | None = None
172    current_list: list[str] | None = None
173    # Multi-line scalar state: block_style is ">" (folded) or "|" (literal)
174    block_style: str | None = None
175    block_parts: list[str] = []
176
177    def _flush_block() -> None:
178        """Finalize a multi-line scalar block and store it in result."""
179        nonlocal block_style, block_parts
180        if current_key is not None and block_style is not None and block_parts:
181            if block_style in (">", ">-"):
182                result[current_key] = " ".join(block_parts)
183            else:  # "|" or "|-"
184                result[current_key] = "\n".join(block_parts)
185        block_style = None
186        block_parts = []
187
188    for line in raw.splitlines():
189        stripped = line.strip()
190        indent = len(line) - len(line.lstrip())
191
192        # If we're collecting a multi-line scalar block, check for continuation
193        if block_style is not None:
194            if indent > 0 and stripped:
195                block_parts.append(stripped)
196                continue
197            else:
198                _flush_block()
199                # Fall through to process this line normally
200
201        # Continuation of a multi-line list (- item form)
202        if (
203            stripped.startswith("- ")
204            and current_key is not None
205            and current_list is not None
206        ):
207            current_list.append(_parse_list_item(stripped[2:].strip()))
208            result[current_key] = current_list
209            continue
210
211        # If we were collecting a list and hit a non-list line, close it
212        if current_list is not None:
213            current_list = None
214
215        # Skip blank lines and comments
216        if not stripped or stripped.startswith("#"):
217            continue
218
219        # key: value pair
220        colon_idx = line.find(":")
221        if colon_idx == -1:
222            continue
223
224        key = line[:colon_idx].strip()
225        value_str = line[colon_idx + 1 :].strip()
226
227        if not key:
228            continue
229
230        # QA-008: warn when an indented sub-key looks like a nested mapping.
231        # Nested mappings are not supported by this parser (see docstring).
232        # Emit a stderr warning so vault-doctor checks don't silently mislead.
233        # This is safe: hook scripts print JSON to stdout; stderr is ignored.
234        if indent > 0 and key:
235            _warning = (
236                f"parse_frontmatter: nested mapping key '{key}' (indented) is not "
237                "supported and will be treated as a top-level scalar. "
238                "Consider flattening this YAML key."
239            )
240            print(_warning, file=sys.stderr)
241            record_parse_warning(_warning)
242
243        current_key = key
244
245        # Empty value -- could be start of a multi-line list
246        if not value_str:
247            current_list = []
248            result[key] = current_list
249            continue
250
251        # Multi-line scalar block indicators: >, |, >-, |-
252        if value_str in (">", "|", ">-", "|-"):
253            block_style = value_str
254            block_parts = []
255            current_list = None
256            continue
257
258        # Inline list: [a, b, c]
259        list_match = _YAML_LIST_INLINE_RE.match(value_str)
260        if list_match:
261            inner = list_match.group(1).strip()
262            if not inner:
263                result[key] = []
264            else:
265                items = [
266                    _parse_list_item(item.strip()) for item in _split_list_items(inner)
267                ]
268                result[key] = items
269            current_list = None
270            continue
271
272        # Scalar value
273        result[key] = _parse_scalar(value_str)
274        current_list = None
275
276    # Flush any remaining block at end of frontmatter
277    _flush_block()
278
279    return result

def atomic_write_text(path: pathlib.Path, content: str, encoding: str = 'utf-8') -> None: 
295def atomic_write_text(path: Path, content: str, encoding: str = "utf-8") -> None:
296    """Write *content* to *path* atomically via a sibling tmp file + ``Path.replace``.
297
298    The tmp file lives in *path*'s own directory so the final ``Path.replace``
299    is a same-filesystem rename -- atomic on POSIX and safe against readers
300    observing a half-written file.  When *path* already exists, its
301    permission bits are copied onto the tmp file before the replace so an
302    existing ``chmod`` is not silently reset.  The tmp file is removed if
303    either the write or the replace fails.  The tmp sibling is created
304    exclusively (``O_CREAT | O_EXCL``, plus ``O_NOFOLLOW`` where available)
305    so a planted ``.tmp`` symlink is never written through (SEC-005).
306
307    Args:
308        path: Destination file path.
309        content: Text content to write.
310        encoding: Text encoding (default: utf-8).
311    """
312    existing_mode: int | None = None
313    try:
314        existing_mode = stat.S_IMODE(os.stat(path).st_mode)
315    except OSError:
316        pass
317
318    tmp = path.parent / (path.name + ".tmp")
319    try:
320        # SEC-005: create the tmp sibling exclusively and never through a
321        # symlink. A planted `X.md.tmp -> ~/.claude/settings.json` used to be
322        # followed by write_text, so the rename clobbered the target. O_EXCL
323        # refuses any existing entry (symlink included); O_NOFOLLOW is kept as
324        # defence in depth (same pattern as session_start/context.py).
325        flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL | getattr(os, "O_NOFOLLOW", 0)
326        try:
327            fd = os.open(tmp, flags, 0o666)
328        except FileExistsError:
329            # Stale tmp from a crashed write, or a planted symlink: unlink
330            # the entry itself (unlink never follows a symlink, so the
331            # target is untouched) and retry exactly once. A second failure
332            # means something is actively racing us — surface it.
333            tmp.unlink()
334            fd = os.open(tmp, flags, 0o666)
335        with os.fdopen(fd, "w", encoding=encoding) as f:
336            f.write(content)
337        if existing_mode is not None:
338            os.chmod(tmp, existing_mode)
339        tmp.replace(path)
340    except OSError:
341        try:
342            tmp.unlink()
343        except OSError:
344            pass
345        raise

482
483    Lowercases the text, replaces spaces and underscores with hyphens,
484    transliterates non-ASCII characters to their closest ASCII equivalent
485    (e.g. "é" -> "e"), removes remaining special characters, and collapses

508                else:
